1"""Per-document download cache for native-markdown external images. 

2 

3Mirrors the ``.mineru_raw`` / ``.docling_raw`` raw-bundle pattern, scoped to the 

4one expensive native-markdown step: downloading external ``http(s)`` images. 

5The bundle lives in a ``<file>.native_raw/`` directory that is a **sibling** of 

6``<file>.parsed/`` so it survives the ``rmtree(parsed_dir)`` that 

7:meth:`NativeParserBase.parse` performs before every re-extraction. 

8 

9Bundle layout:: 

10 

11 <file>.native_raw/ 

12 _manifest.json # atomic success marker + cache key 

13 <sha256(url)[:16]>.png # one file per cached image (final bytes) 

14 ... 

15 

16The manifest is the cache key: a bundle is reused only when the source file 

17content hash AND the download-options signature both still match, mirroring the 

18external engines. On a hit the resolver reuses the stored bytes (already 

19post-SVG-rasterization), skipping both the network fetch and the rasterization. 

20""" 

21

40def native_md_options_signature() -> str: 

41 """A ``sha256`` over the download knobs that change an image's bytes. 

42 

43 Deliberately excludes ``NATIVE_MD_IMAGE_DOWNLOAD_ENABLED`` / 

44 ``..._TIMEOUT`` / ``..._REQUIRED`` — those gate *whether* a fetch happens, 

45 not the resulting bytes — and includes the size / SVG-pixel ceilings and the 

46 SSRF allowlist (which govern what bytes are accepted at all). 

47 

48 The per-document budgets (``..._MAX_TOTAL_BYTES`` / ``..._MAX_REQUESTS`` / 

49 ``..._DOWNLOAD_TOTAL_TIMEOUT``) are excluded for the same reason: they 

50 decide whether a fetch is attempted, never what a given URL yields. Two 

51 further arguments against including them. A budget-rejected image is never 

52 ``put()`` here, so there is no entry to go stale, and the byte budget is 

53 re-applied to cache HITS at resolve time, so tightening it takes effect on 

54 the next parse with no invalidation. And including them would make tuning 

55 any budget invalidate every ``.native_raw/`` bundle on the host at once, 

56 triggering exactly the re-download storm the budgets exist to prevent.""" 

57 payload = { 

58 "signature_version": 1, 

59 "max_bytes": os.getenv("NATIVE_MD_IMAGE_MAX_BYTES", ""), 

60 "max_svg_pixels": os.getenv("NATIVE_MD_IMAGE_MAX_SVG_PIXELS", ""), 

61 "allowed_non_public_cidrs": os.getenv( 

62 "NATIVE_MD_IMAGE_ALLOWED_NON_PUBLIC_CIDRS", "" 

63 ), 

64 } 

65 raw = json.dumps(payload, sort_keys=True, separators=(",", ":")) 

66 return "sha256:" + hashlib.sha256(raw.encode("utf-8")).hexdigest()

69def _url_filename(url: str, fmt: str) -> str: 

70 digest = hashlib.sha256(url.encode("utf-8")).hexdigest()[:16] 

71 ext = fmt or "bin" 

72 return f"{digest}.{ext}"

75class NativeImageRawCache: 

76 """Reuse already-downloaded external images across re-parses. 

77 

78 Per-document and single-writer: each parse owns one ``raw_dir`` and the 

79 parse queue serializes work per document, so no locking is needed. 

80 """ 

81 

82 def __init__( 

83 self, 

84 raw_dir: Path, 

85 *, 

86 source_path: Path, 

87 options_signature: str, 

88 force_reparse: bool, 

89 ) -> None: 

90 self._raw_dir = raw_dir 

91 self._source_path = source_path 

92 self._options_signature = options_signature 

93 self._force_reparse = force_reparse 

94 self._source_hash = "" 

95 self._valid = False 

96 self._cleared = False 

97 # ``_dirty`` gates the manifest write: it is set only when a real 

98 # download is stored (:meth:`put`). A run that only reuses cached images 

99 # (a pure hit) leaves it ``False`` so :meth:`flush` touches nothing on 

100 # disk — the bundle's mtimes then reveal whether the cache was hit. 

101 self._dirty = False 

102 # Index of reusable entries from a valid prior bundle (url -> entry). 

103 self._index: dict[str, dict] = {} 

104 # Entries referenced this run (reused or freshly put) -> manifest output. 

105 self._entries: dict[str, dict] = {}

107 def load(self) -> None: 

108 """Compute the current source hash and decide whether the on-disk 

109 bundle is a cache hit (valid) or must be rebuilt.""" 

110 try:

112 except OSError as exc:

114 self._source_hash = "" 

115 if self._force_reparse: 

116 return 

117 manifest = self._read_manifest() 

118 if manifest is None: 

119 return 

120 if ( 

121 manifest.get("source_content_hash") != self._source_hash 

122 or manifest.get("options_signature") != self._options_signature 

123 ): 

124 return

127 return

156 def put(self, url: str, data: bytes, fmt: str) -> None: 

157 """Store freshly-downloaded image bytes and record them for the manifest.""" 

158 self._ensure_writable_dir() 

159 file_name = _url_filename(url, fmt) 

160 try: 

161 (self._raw_dir / file_name).write_bytes(data) 

162 except OSError as exc: 

163 logger.warning("[native_md_cache] failed to write cache file: %s", exc) 

164 return 

165 self._entries[url] = { 

166 "file": file_name, 

167 "sha256": "sha256:" + hashlib.sha256(data).hexdigest(), 

168 "size": len(data), 

169 "fmt": fmt, 

170 } 

171 self._dirty = True

173 def flush(self, *, prune: bool = True) -> None: 

174 """Persist the bundle only if it actually changed this run. 

175 

176 Skipped entirely on a **pure cache hit** (every image reused, nothing 

177 downloaded) — the source is byte-identical, so the referenced image set 

178 equals the on-disk one and a rewrite would only be an idempotent write. 

179 Leaving the manifest and image files untouched means their mtimes flag a 

180 miss/update vs. a hit. Also a no-op for an image-less or 

181 download-disabled run, so a pre-existing valid bundle is left intact. 

182 

183 ``prune=False`` is the failure path: a parse that raises after 

184 downloading images still persists that progress, so the retry resumes 

185 from the cache instead of re-fetching. Pruning must be skipped there — 

186 it keys on the entries referenced THIS run, and an aborted run never 

187 reached the rest — so the manifest instead keeps every entry from the 

188 prior valid bundle and no files are deleted.""" 

189 if not self._dirty: 

190 return 

191 self._ensure_writable_dir() 

192 if prune: 

193 entries = self._entries 

194 referenced = {e["file"] for e in entries.values()} 

195 for child in self._raw_dir.iterdir(): 

196 if child.name == _MANIFEST_FILENAME: 

197 continue 

198 if child.is_file() and child.name not in referenced: 198 ↛ 199line 198 didn't jump to line 199 because the condition on line 198 was never true

199 try: 

200 child.unlink() 

201 except OSError: 

202 pass 

203 else: 

204 # _index is non-empty only when the prior bundle was valid, in 

205 # which case no clear ran and its files are still on disk. 

206 entries = {**self._index, **self._entries} 

207 manifest = { 

208 "version": _MANIFEST_VERSION, 

209 "engine": _CACHE_ENGINE, 

210 "source_content_hash": self._source_hash, 

211 "options_signature": self._options_signature, 

212 "images": entries, 

213 } 

214 final = self._raw_dir / _MANIFEST_FILENAME 

215 tmp = final.with_suffix(".json.tmp") 

216 try: 

217 tmp.write_text( 

218 json.dumps(manifest, ensure_ascii=False, indent=2), encoding="utf-8" 

219 ) 

220 os.replace(tmp, final) 

221 except OSError as exc: 

222 logger.warning("[native_md_cache] failed to write manifest: %s", exc)

224 def _ensure_writable_dir(self) -> None: 

225 """Create ``raw_dir`` and, on the first write of an invalidated bundle, 

226 drop the stale contents so reused entries never mingle with old ones.""" 

227 self._raw_dir.mkdir(parents=True, exist_ok=True) 

228 if not self._valid and not self._cleared: 

229 clear_dir_contents(self._raw_dir) 

230 self._cleared = True
