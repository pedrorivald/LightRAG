1"""Shared zip-bundle extraction for external parser engines. 

2 

3Engines like docling return their full output as a zip archive. This helper 

4extracts it safely (refusing path traversal / absolute paths) into a target 

5directory. Engine-specific post-extraction normalization (e.g. mineru's 

6nested-subdir hoist) is *not* done here — each engine's client handles its 

7own quirks. 

8""" 

9 

10from __future__ import annotations 

11 

12import io 

13import os 

14import zipfile 

15from pathlib import Path 

16 

17from lightrag.constants import ( 

18 DEFAULT_PARSER_RESULT_BUNDLE_MAX_ENTRIES, 

19 DEFAULT_PARSER_RESULT_BUNDLE_MAX_TOTAL_BYTES, 

20) 

21from lightrag.utils import get_env_value 

22 

23 

24def result_bundle_limits() -> tuple[int | None, int | None]: 

25 """Live-read the external-engine result-bundle zip-bomb budget. 

26 

27 Returns ``(max_entries, max_total_bytes)`` for :func:`safe_extract_zip`, 

28 read from the environment at call time so an operator whose legitimate 

29 bundle is refused can raise the ceiling without a code change. A 

30 non-positive value maps to ``None`` (that gate disabled) — ``safe_extract_zip`` 

31 treats ``None`` as unlimited, so passing 0 through raw would refuse every 

32 bundle instead of disabling the check. 

33 """ 

34 max_entries = get_env_value( 

35 "PARSER_RESULT_BUNDLE_MAX_ENTRIES", 

36 DEFAULT_PARSER_RESULT_BUNDLE_MAX_ENTRIES, 

37 int, 

38 ) 

39 max_total_bytes = get_env_value( 

40 "PARSER_RESULT_BUNDLE_MAX_TOTAL_BYTES", 

41 DEFAULT_PARSER_RESULT_BUNDLE_MAX_TOTAL_BYTES, 

42 int, 

43 ) 

44 return ( 

45 max_entries if max_entries > 0 else None, 

46 max_total_bytes if max_total_bytes > 0 else None, 

47 ) 

48 

49 

50def safe_extract_zip( 

51 payload: bytes, 

52 dest_dir: Path, 

53 *, 

54 max_entries: int | None = None, 

55 max_total_bytes: int | None = None, 

56) -> list[str]: 

57 """Extract a zip archive into ``dest_dir``, refusing unsafe paths. 

58 

59 Raises ``RuntimeError`` if any entry name is absolute or contains ``..`` 

60 components after normalization. Returns the list of extracted member 

61 names (as stored in the zip, prior to OS-specific normalization), so 

62 callers can validate the bundle layout without re-walking the directory. 

63 

64 Optional zip-bomb guards (both default ``None`` = unlimited, preserving the 

65 original behaviour for existing callers): ``max_entries`` caps the member 

66 count and ``max_total_bytes`` caps the summed *uncompressed* size declared 

67 in the archive's central directory. Both are checked from ``infolist()`` 

68 metadata *before* extraction, so a malicious archive is rejected without 

69 being written to disk. 

70 """ 

71 dest_dir.mkdir(parents=True, exist_ok=True) 

72 buf = io.BytesIO(payload) 

73 with zipfile.ZipFile(buf) as zf: 

74 infos = zf.infolist() 

75 if max_entries is not None and len(infos) > max_entries: 

76 raise RuntimeError( 

77 f"Refusing zip with {len(infos)} entries (max {max_entries})" 

78 ) 

79 if max_total_bytes is not None: 

80 total = sum(info.file_size for info in infos) 

81 if total > max_total_bytes: 

82 raise RuntimeError( 

83 f"Refusing zip: uncompressed size {total} bytes " 

84 f"exceeds limit {max_total_bytes}" 

85 ) 

86 names = zf.namelist() 

87 for name in names: 

88 norm = os.path.normpath(name) 

89 if ( 

90 norm.startswith("..") 

91 or os.path.isabs(norm) 

92 or norm.startswith(("/", os.sep)) 

93 ): 

94 raise RuntimeError(f"Refusing zip entry with unsafe path: {name!r}") 

95 zf.extractall(dest_dir) 

96 return names 

97 

98 

99__all__ = ["result_bundle_limits", "safe_extract_zip"]
