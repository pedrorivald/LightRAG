1"""Bounded in-process metrics for the scheduling paths (LR2 Phase 6, item 3). 

2 

3The bounded-memory rework moved several decisions off the happy path — a paged 

4sweep, a strict active count per admission, a drain-then-reset freeze — and each 

5one can be slow or noisy without anything visible going wrong. These counters and 

6duration summaries are the only way to tell "the sweep is paging" from "the sweep 

7is thrashing", or "a freeze is normal" from "a freeze is stuck". 

8 

9Design constraints that shaped this module: 

10 

11* **Fixed key set.** Every metric name is declared below. An unknown name is 

12 logged once and dropped rather than inserted — a metrics registry that grows 

13 with cardinality is exactly the unbounded structure this whole plan exists to 

14 remove. 

15* **O(1) per metric.** Durations keep count / total / max / last, not samples. 

16 There are no percentiles: a ring of samples would be a per-metric bound to 

17 pick, defend and test, and the questions above are answered by max plus mean. 

18* **In-process, per worker.** Incrementing a shared (Manager-hosted) counter 

19 would add an RPC to paths taken thousands of times per sweep, which is the cost 

20 the scheduling rework was built to avoid. Under gunicorn with N workers 

21 ``/health`` therefore reports the numbers of the worker that answered — the 

22 same documented boundary as the login rate limiter and the keyed-lock table. 

23 The pipeline supervisor is single-owner (whoever holds ``busy``), so its 

24 measurements are not spread across workers in the first place. 

25* **Never fatal.** A metrics failure must not fail the operation it measures; 

26 every entry point swallows its own errors. 

27 

28Deliberately NOT duplicated here: scan classification counts, source-conflict 

29sightings and abandoned scan jobs. Those already have a bounded, per-job surface 

30in the scan job record (``GET /documents/scan/status/{track_id}``) — counting 

31them twice would invite the two numbers to disagree. 

32""" 

33 

34from __future__ import annotations 

35 

36import threading 

37from typing import Any, Dict 

38 

39from lightrag.utils import logger 

40 

41# --- counters ------------------------------------------------------------- 

42# Ingress refused because a manual retry froze new work (LR2 §6.1/§7.2). A 

43# handful during a reset is normal; a steadily climbing number means clients are 

44# retrying into a freeze that is not ending. 

45FREEZE_REJECTS = "freeze_rejects" 

46# Auto-rescan dirty flag re-armed: a processing request that could not name doc 

47# ids (busy-refused, partial commit, recovery). Each one costs a later strict 

48# sweep, so a high rate explains sweep load that no upload seems to account for. 

49AUTO_RESCAN_REARMS = "auto_rescan_rearms" 

50# Documents moved FAILED→PENDING by an exclusive reset, and the pages it took. 

51MANUAL_RESET_DOCS = "manual_reset_docs" 

52MANUAL_RESET_PAGES = "manual_reset_pages" 

53 

54_COUNTERS: tuple[str, ...] = ( 

55 FREEZE_REJECTS, 

56 AUTO_RESCAN_REARMS, 

57 MANUAL_RESET_DOCS, 

58 MANUAL_RESET_PAGES, 

59) 

60 

61# --- duration summaries --------------------------------------------------- 

62# One keyset page of the doc_status sweep (query + projection). 

63SCHEDULING_PAGE_SECONDS = "scheduling_page_seconds" 

64# One strict active-document count, taken per admission decision. 

65ACTIVE_COUNT_SECONDS = "active_count_seconds" 

66# DRAIN_TO_IDLE: from freezing ingress to confirmed quiescence. This is the 

67# window during which uploads are refused, so it is the one an operator feels. 

68MANUAL_DRAIN_SECONDS = "manual_drain_seconds" 

69# EXCLUSIVE_RESET: the paged FAILED→PENDING rewrite, with no worker running. 

70MANUAL_RESET_SECONDS = "manual_reset_seconds" 

71 

72_DURATIONS: tuple[str, ...] = ( 

73 SCHEDULING_PAGE_SECONDS, 

74 ACTIVE_COUNT_SECONDS, 

75 MANUAL_DRAIN_SECONDS, 

76 MANUAL_RESET_SECONDS, 

77) 

78 

79_lock = threading.Lock() 

80_counters: Dict[str, int] = {name: 0 for name in _COUNTERS} 

81_durations: Dict[str, Dict[str, float]] = { 

82 name: {"count": 0, "total_seconds": 0.0, "max_seconds": 0.0, "last_seconds": 0.0} 

83 for name in _DURATIONS 

84} 

85 

86 

87def increment(name: str, amount: int = 1) -> None: 

88 """Add to a declared counter; unknown names are dropped, never inserted.""" 

89 try: 

90 if name not in _counters: 

91 logger.warning(f"Ignoring unknown pipeline metric counter {name!r}") 

92 return 

93 if amount <= 0: 

94 return 

95 with _lock: 

96 _counters[name] += amount 

97 except Exception as metric_error: # pragma: no cover - defensive 

98 logger.debug(f"pipeline metric {name!r} not recorded: {metric_error}") 

99 

100 

101def observe(name: str, seconds: float) -> None: 

102 """Record one duration sample into a declared summary.""" 

103 try: 

104 if name not in _durations: 

105 logger.warning(f"Ignoring unknown pipeline metric duration {name!r}") 

106 return 

107 value = max(0.0, float(seconds)) 

108 with _lock: 

109 summary = _durations[name] 

110 summary["count"] += 1 

111 summary["total_seconds"] += value 

112 summary["last_seconds"] = value 

113 if value > summary["max_seconds"]: 

114 summary["max_seconds"] = value 

115 except Exception as metric_error: # pragma: no cover - defensive 

116 logger.debug(f"pipeline metric {name!r} not recorded: {metric_error}") 

117 

118 

119def snapshot() -> Dict[str, Any]: 

120 """Current values, rounded for display. Bounded by the declared key set.""" 

121 with _lock: 

122 counters = dict(_counters) 

123 durations = { 

124 name: { 

125 "count": int(summary["count"]), 

126 "total_seconds": round(summary["total_seconds"], 4), 

127 "max_seconds": round(summary["max_seconds"], 4), 

128 "last_seconds": round(summary["last_seconds"], 4), 

129 "mean_seconds": ( 

130 round(summary["total_seconds"] / summary["count"], 4) 

131 if summary["count"] 

132 else 0.0 

133 ), 

134 } 

135 for name, summary in _durations.items() 

136 } 

137 return {"counters": counters, "durations": durations} 

138 

139 

140def reset() -> None: 

141 """Zero everything (tests only; there is no runtime reset endpoint).""" 

142 with _lock: 

143 for name in _counters: 

144 _counters[name] = 0 

145 for summary in _durations.values(): 

146 summary.update( 

147 { 

148 "count": 0, 

149 "total_seconds": 0.0, 

150 "max_seconds": 0.0, 

151 "last_seconds": 0.0, 

152 } 

153 )
