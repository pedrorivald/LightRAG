125def truncate_to_heading_length(text: str) -> str: 

126 """Bound a synthesized heading to BOTH the strong-body weighted cap 

127 (:func:`heading_max_chars`) AND the hard raw ceiling ``MAX_HEADING_LENGTH``, 

128 appending ``...`` when either is exceeded. The raw ceiling keeps 

129 ``parse_document.truncate_heading`` a no-op so the four landing sites of a 

130 title block (H1 / doc_title / first_heading / every descendant's 

131 parent_headings root) never split, even when the env cap is set above 200.""" 

132 cap = heading_max_chars() 

133 if weighted_char_length(text) <= cap and len(text) <= MAX_HEADING_LENGTH: 

134 return text 

135 w_budget = max(cap - 3, 0) # reserve the "..." (weight 3) 

136 r_budget = MAX_HEADING_LENGTH - 3 

137 used_w = used_r = 0 

138 out: list[str] = [] 

139 for ch in text: 139 ↛ 148line 139 didn't jump to line 148 because the loop on line 139 didn't complete

140 w = 3 if "一" <= ch <= "鿿" else 1 

141 if used_w + w > w_budget or used_r + 1 > r_budget: 

142 break 

143 out.append(ch) 

144 used_w += w 

145 used_r += 1 

146 # rstrip so a cut landing right after the " " separator does not leave 

147 # "词 ..." with a dangling double space. 

148 return "".join(out).rstrip() + "..."

156def _strip_leading_numbering(text: str) -> str: 

157 """Text with a leading numbering label removed ("3.1.1 X" → "X"). 

158 

159 A numbering label is structural, not prose. Fed to spaCy its dots 

160 mis-segment a numbered heading into pseudo-sentences ("3.1.1 桌面…" → 

161 "3.1." | "1 桌面…"), so a legitimate numbered heading falsely reads as ≥2 

162 sentences. Reuses the numbering machinery (:func:`classify_numbering`) as 

163 the single source of what the label is; a no-op when there is none. 

164 """ 

165 cls = classify_numbering(text) 

166 if cls is None or not cls.label_text: 

167 return text 

168 head = text.lstrip() 

169 if head.startswith(cls.label_text): 169 ↛ 175line 169 didn't jump to line 175 because the condition on line 169 was always true

170 # Drop the label plus its trailing separator run (". 、:）" etc.). 

171 # \s (not a literal-space whitelist) so NBSP/full-width spaces go too — 

172 # a leading \xa0 survives into the prose otherwise, and zh spaCy counts 

173 # it as its own "sentence", falsely demoting headings such as 第二章. 

174 return re.sub(r"^[\s、.。:：)）]+", "", head[len(cls.label_text) :]) 

175 return text

204def strong_body_reason(text: str) -> str | None: 

205 """Return the rule id when ``text`` carries a strong body feature. 

206 

207 Rules (any one suffices): 

208 - ``imprint_marker``: opens with a 公文版记 (imprint) ANCHOR — a colon 

209 prefix (抄送：/ 主题词：) — checked first, before any length / spaCy 

210 rule, so an anchor line keeps its imprint identity however long it 

211 runs. (The 印发-family CLOSER is region-scoped and deliberately NOT 

212 part of strong_body; see :func:`imprint_closer_reason`.); 

213 - ``strong_body_length``: weighted length > 180 en-equivalent chars; 

214 - ``strong_body_multi_sentence``: spaCy sees ≥2 sentences AND a visible 

215 internal sentence terminator corroborates the split — the spaCy vote 

216 alone is not trusted (the pinned zh model hallucinates mid-word breaks 

217 on short title fragments: 索菲|亚, 承诺|书); 

218 - ``strong_body_sentence_end``: ends with a sentence terminator 

219 (stepping over closing quotes/brackets); a trailing English period 

220 counts only when spaCy segmentation says it closes a sentence. 

221 

222 Returns None when no rule fires (the paragraph MAY be a heading). 

223 """

225 if reason is not None: 

226 return reason 

227 stripped = text.strip() 

228 if not stripped: 228 ↛ 229line 228 didn't jump to line 229 because the condition on line 228 was never true

229 return None 

230 if weighted_char_length(stripped) > heading_max_chars(): 

231 return "strong_body_length" 

232 

233 tail = stripped 

234 while tail and tail[-1] in _CLOSING_WRAPPERS: 234 ↛ 235line 234 didn't jump to line 235 because the condition on line 234 was never true

235 tail = tail[:-1].rstrip() 

236 if tail and tail[-1] in SENTENCE_END_CHARS: 236 ↛ 238line 236 didn't jump to line 238 because the condition on line 236 was always true

237 return "strong_body_sentence_end" 

238 if tail.endswith(".") and nlp.ends_with_sentence_period(tail): 

239 return "strong_body_sentence_end"

241 # Judge sentence shape over the title PROSE, not a leading numbering label: 

242 # spaCy splits multi-level numbers ("3.1.1") into pseudo-sentences, which 

243 # would falsely demote every dotted-numbered heading to body. 

244 # 

245 # The spaCy ≥2 vote is NECESSARY but not SUFFICIENT: the pinned zh model's 

246 # dependency parser hallucinates sentence boundaries mid-word on short 

247 # heading fragments (广州市增城区“7.19”索菲亚… splits at 索菲|亚, 承诺书 at 

248 # 承诺|书), and a bigger model just relocates the errors (zh_core_web_lg is 

249 # no better than _sm). So a DETERMINISTIC corroborator CONSTRAINS (does not 

250 # replace) the model vote: still require spaCy's ≥2, but trust it only when 

251 # a visible internal sentence terminator (。！？…；;.!?, decimals excluded) 

252 # corroborates the split — a mid-word pseudo-split never produces one. 

253 # Known tradeoff: punctuation-less multi-clause prose (e.g. a lead-in ending 

254 # in “：”) also loses the verdict; accepted as body-vs-heading rarely hinges 

255 # on it and length / sentence-end rules still cover genuine long prose. 

256 prose = _strip_leading_numbering(stripped)

258 prose 

259 and nlp.sentence_count(prose) >= 2 

260 and has_explicit_internal_sentence_boundary(stripped) 

261 ):

431def imprint_closer_reason(text: str) -> str | None: 

432 """A 版记 region CLOSER line — 印发-family — is metadata. 

433 

434 Deliberately NOT a standalone per-line rule and NOT part of 

435 :func:`strong_body_reason`: a line-final ``印发`` occurs in body prose 

436 (…已印发) too, so callers invoke this ONLY inside the forward window of an 

437 imprint anchor (:func:`title_block.detect_imprint_regions`). That scoping 

438 is exactly how the "印发 must follow 抄送/主题词" constraint is enforced. 

439 

440 Two forms (env-tunable, comma/pipe-separated): 

441 - prefix class (印发 / 印发机关): the line OPENS with the marker + a colon 

442 or any whitespace (印发：XX办公厅 / 印发 XX办公厅 / 印发机关　XX办公厅). ``\\s`` 

443 includes a soft line break and the ideographic space, so a two-line 

444 ``印发机关\\n机构名`` still hits; 

445 - trailing class (印发): the line ENDS with the marker (某某办公室 

446 2026年6月30日 印发) — the GB/T layout with the issuer/date first. A 

447 trailing period stops the match (已印发。 does not close a region). 

448 

449 Operates on the RAW text: the leading/trailing whitespace is evidence. 

450 """ 

451 head = text.lstrip() 

452 if not head: 

453 return None 

454 for prefix in _env_items( 

455 "DOCX_SMART_IMPRINT_CLOSER_PREFIXES", 

456 DEFAULT_DOCX_SMART_IMPRINT_CLOSER_PREFIXES, 

457 ): 

458 body = _IMPRINT_GAP.join(re.escape(ch) for ch in prefix) 

459 # Prefix marker followed by a (optionally gap-padded) colon OR any 

460 # whitespace — mirrors the anchor tails but for the 印发 opener. 

461 if re.match(body + r"(?:" + _IMPRINT_GAP + r"[：:]|\s)", head): 

462 return "imprint_closer" 

463 tail = text.rstrip() 

464 for marker in _env_items( 

465 "DOCX_SMART_IMPRINT_CLOSER_TRAILING", 

466 DEFAULT_DOCX_SMART_IMPRINT_CLOSER_TRAILING, 

467 ): 

468 body = _IMPRINT_GAP.join(re.escape(ch) for ch in marker) 

469 m = re.search(body + r"$", tail) 

470 # The GB/T trailing form carries the issuer/date BEFORE 印发, so a bare 

471 # label (印发 alone) must not fire — require something to precede it. 

472 if m is not None and m.start() > 0: 

473 return "imprint_closer" 

474 return None

479#: 2026-7-31; the separator must be consistent within one date). Zero is 

480#: written 〇 / ○ (circle) / 零 in practice. 

481_CN_DATE_DIGIT = "〇○零一二三四五六七八九十两廿" 

482_DOCUMENT_DATE_CN = re.compile( 

483 rf"^[{_CN_DATE_DIGIT}]{{2,4}}年[{_CN_DATE_DIGIT}]{{1,3}}月[{_CN_DATE_DIGIT}]{{1,3}}日$" 

484) 

485_DOCUMENT_DATE_AR = re.compile(r"^\d{2,4}\s*年\s*\d{1,2}\s*月\s*\d{1,2}\s*日$") 

486_DOCUMENT_DATE_SEP = re.compile(r"^\d{4}([./\-])\d{1,2}\1\d{1,2}$")

489def is_document_date(text: str) -> bool: 

490 """True when the WHOLE line is a 成文日期 (a bare document date), e.g. 

491 ``二○○九年七月六日``, ``2009年7月6日``, ``2026.7.31``, ``2026/7/31`` or 

492 ``2026-7-31``. 

493 

494 A line that merely CONTAINS a date (第十六条…自2009年7月1日起施行。) is NOT a 

495 document date — the ``^…$`` anchors demand the date be the entire line. 

496 Two consumers: pulling a mis-ordered 成文日期 trailing a 版记 into the

498 date line from the solo centered-heading channel — a 

499 centered 落款 date must never read as a heading. 

500 """ 

501 s = text.strip() 

502 return bool( 

503 _DOCUMENT_DATE_CN.match(s) 

504 or _DOCUMENT_DATE_AR.match(s) 

505 or _DOCUMENT_DATE_SEP.match(s) 

506 )

509#: Net leading whitespace (in CJK em — see ``features.leading_pad_em``) a 

510#: ``w:jc=center`` paragraph may carry and still count as visually centered. 

511#: Word centers the line INCLUDING its leading padding, so the visible text is 

512#: pushed right by HALF this budget: at 4.0 em the text sits ~2 characters right 

513#: of true center, which still reads as centered; the 落款 shape this guards 

514#: against overshoots by an order of magnitude (the observed 公函 signature line 

515#: pads 29 spaces = 14.5 em, i.e. ~7 characters of shift). 

516#: 

517#: Measured over 32 documents / 243 centered paragraphs (real 公文 corpus plus 

518#: every golden fixture): 240 paragraphs pad 0.0 em, one 1.0 em, and the two 

519#: signature lines this rule targets 14.5/15.0 em — so any threshold in 

520#: (1.0, 14.5] is equivalent on that corpus and 4.0 sits deliberately near the 

521#: conservative end. Tune here. 

522CENTER_MAX_LEADING_PAD_EM = 4.0

525def is_visually_centered(rec: Any) -> bool: 

526 """True when ``rec`` is centered AND renders that way. 

527 

528 ``w:jc=center`` alone is not enough. 公文 authors routinely produce a 

529 右下角落款/署名 by centering a paragraph and then padding it with a run of 

530 spaces, which shifts the visible text right while the alignment attribute 

531 still says "center". Such a line is shape-identical to a real same-size 

532 centered title, so the solo centered-heading channel promoted it — the 

533 residue documented on ``heading_flow.gate_candidates``' centered channel. 

534 

535 Drop-in replacement for ``rec.alignment == "center"``, and BOTH consumers 

536 must use it: the candidate gate (admission) and the leveling band split 

537 (``centered_plain`` vs ``uncentered_plain``). Using it in only one place 

538 splits the definition of "centered" between the two. 

539 

540 Losing centeredness is NOT a veto: outline / numbering / font-size / bold 

541 paths stay open, exactly as for the anti-poetry run cap. 

542 """ 

543 if rec.alignment != "center": 

544 return False 

545 return rec.leading_pad_em < CENTER_MAX_LEADING_PAD_EM

549 """True when the line carries no letter at all — page numbers (``- 1 -``),

590def canonicalize_paragraph_text(text: str) -> str: 

591 """Whitespace-free, markdown-prefix-free canonical form (I1 comparisons). 

592 

593 ALL whitespace goes (the sidecar-backfill lesson: collapsing to single 

594 spaces broke zero-width CJK splits), and a leading markdown ``#`` run is 

595 stripped so rendered heading lines compare equal to their source 

596 paragraphs. Placeholder tokens are left as-is — table/equation/drawing 

597 payloads are covered by their own asset integrity checks, not I1. 

598 """ 

599 stripped = _HEADING_PREFIX_RE.sub("", (text or "").strip()) 

600 return re.sub(r"\s+", "", stripped)

611def _detect_plain_numbered_toc( 

612 records: Sequence[Any], min_lines: int 

613) -> tuple[set[int], list[dict]]: 

614 """Third TOC channel: a leaderless run of short numbered lines under a 

615 standalone 目次 / 目录 / (Table of) Contents title, confirmed by every line 

616 having a duplicate later in the body (a GB/T-style TOC typed as plain text 

617 with no dot leaders). See the four safety constraints below — this channel 

618 deletes real content (``toc_indices`` is the I1 whitelist), so it errs 

619 toward false negatives (a short/unusual TOC is left as candidates, which 

620 CB1 handles) rather than deleting body. 

621 """ 

622 # Line-level inverted index: canon(line).casefold() -> record indices where 

623 # it appears. Line-level (not paragraph-level) so soft-broken TOCs match, 

624 # and casefold so an English TOC entry matches a differently-cased heading. 

625 canon_at: dict[str, list[int]] = {} 

626 for i, rec in enumerate(records): 

627 if rec.kind != "para": 

628 continue 

629 for ln in _visible_lines(rec): 

630 key = canonicalize_paragraph_text(ln).casefold() 

631 if key: 631 ↛ 629line 631 didn't jump to line 629 because the condition on line 631 was always true

632 canon_at.setdefault(key, []).append(i) 

633 

634 def _copy_after(line: str, after: int) -> bool: 

635 for variant in (line, _TOC_TRAILING_PAGENO_RE.sub("", line)): 

636 key = canonicalize_paragraph_text(variant).casefold() 

637 if key and any(idx > after for idx in canon_at.get(key, ())): 

638 return True 

639 return False 

640 

641 cap = heading_max_chars() 

642 toc: set[int] = set() 

643 events: list[dict] = [] 

644 long_paras_seen = 0 

645 n = len(records) 

646 i = 0 

647 while i < n: 

648 rec = records[i] 

649 if rec.kind == "para": 

650 lines = _visible_lines(rec) 

651 is_long = any(weighted_char_length(ln) > cap for ln in lines) 

652 if is_long: 

653 long_paras_seen += 1 

654 is_anchor = ( 

655 len(lines) == 1 

656 and _TOC_TITLE_RE.match(lines[0]) is not None 

657 and long_paras_seen < _TOC_ANCHOR_MAX_LONG_PARAS_BEFORE 

658 ) 

659 if not is_anchor: 

660 i += 1 

661 continue 

662 

663 # Collect the maximal run of short, forward-duplicated members after 

664 # the anchor. Stop at the first paragraph that breaks any rule — 

665 # incrementally, so the body's first heading (whose only copy is the 

666 # TOC entry ABOVE it) ends the run instead of being swallowed. 

667 block: list[int] = [] 

668 block_lines = 0 

669 probe_sizes: list[float] = [] 

670 toc_max_size: float | None = None 

671 j = i + 1 

672 while j < n: 672 ↛ 707line 672 didn't jump to line 707 because the condition on line 672 was always true

673 r = records[j] 

674 if r.kind == "empty_para": 674 ↛ 675line 674 didn't jump to line 675 because the condition on line 674 was never true

675 j += 1 

676 continue 

677 if r.kind != "para": 677 ↛ 678line 677 didn't jump to line 678 because the condition on line 677 was never true

678 break 

679 mlines = _visible_lines(r) 

680 if not mlines or any(weighted_char_length(ln) > cap for ln in mlines): 

681 break 

682 # Font-size termination: a member strictly larger than the TOC's 

683 # established max size is the body's first heading. The max is 

684 # locked from the first probe MEMBERS' known sizes; if all of 

685 # those are None, size termination stays disabled (other rules 

686 # gate). 

687 size = effective_font_size_pt(r) 

688 if ( 

689 toc_max_size is not None 

690 and size is not None 

691 and size > toc_max_size 

692 ): 

693 break 

694 if not all(_copy_after(ln, j) for ln in mlines): 

695 break 

696 block.append(j) 

697 block_lines += len(mlines) 

698 if toc_max_size is None and len(block) <= _TOC_SIZE_PROBE_MEMBERS: 698 ↛ 703line 698 didn't jump to line 703 because the condition on line 698 was always true

699 if size is not None: 699 ↛ 701line 699 didn't jump to line 701 because the condition on line 699 was always true

700 probe_sizes.append(size) 

701 if len(block) == _TOC_SIZE_PROBE_MEMBERS: 

702 toc_max_size = max(probe_sizes) if probe_sizes else None 

703 j += 1 

704 

705 # all-or-nothing: every member line must have a copy strictly after 

706 # the whole block, and the block must clear the line threshold. 

707 end = block[-1] if block else i 

708 confirmed = block_lines >= min_lines and all( 

709 _copy_after(ln, end) for k in block for ln in _visible_lines(records[k]) 

710 ) 

711 if confirmed: 

712 toc.update(block) 

713 events.append( 

714 { 

715 "rule": "toc_plain_numbered_run", 

716 "anchor": i, 

717 "start": block[0], 

718 "end": end, 

719 } 

720 ) 

721 i = end + 1 

722 continue 

723 i += 1 

724 return toc, events

727def detect_toc_records(records: Sequence[Any]) -> tuple[set[int], list[dict]]: 

728 """Three-channel TOC detection. Returns ``(indices, events)`` — the 

729 record indices to remove from smart output, and third-channel audit events. 

730 

731 1. Structural: TOC field instructions / _Toc bookmark links on a paragraph. 

732 2. Dot-leader run: ≥ DOCX_SMART_TOC_MIN_LINES consecutive leader LINES 

733 ending in a dot leader + page number. Lines are counted, not paragraphs: 

734 the run spans standalone paragraphs AND the soft-break lines inside one 

735 paragraph, so a TOC typed as a single multi-line paragraph is caught 

736 too. An isolated single line is never evidence. 

737 3. Plain numbered run: a leaderless run of short numbered lines under a 

738 目次 / 目录 / (Table of) Contents title, every line duplicated later in 

739 the body (see :func:`_detect_plain_numbered_toc`). 

740 

741 Detection only — the ``smart_toc_removed_paragraphs`` warning is a 

742 content claim, so the caller records it once the smart output (which 

743 actually drops these records) is accepted, not here. 

744 """ 

745 min_lines = _env_int("DOCX_SMART_TOC_MIN_LINES", DEFAULT_DOCX_SMART_TOC_MIN_LINES) 

746 toc: set[int] = set() 

747 for i, rec in enumerate(records): 

748 if rec.kind == "para" and (rec.is_toc_field or rec.is_toc_link): 

749 toc.add(i) 

750 

751 run: list[int] = [] 

752 run_lines = 0 

753 

754 def _flush_run() -> None: 

755 nonlocal run, run_lines 

756 if run_lines >= min_lines: 

757 toc.update(run) 

758 run = [] 

759 run_lines = 0 

760 

761 for i, rec in enumerate(records): 

762 if rec.kind == "empty_para": 

763 continue # blank lines do not break a leader run 

764 if rec.kind == "para": 

765 lines = [ln.strip() for ln in rec.text.split("\n") if ln.strip()] 

766 # A paragraph joins the run only if EVERY visible line is a dot 

767 # leader; its soft-break lines each count toward the line total. 

768 if lines and all(_TOC_DOT_LEADER_RE.search(ln) for ln in lines): 

769 run.append(i) 

770 run_lines += len(lines) 

771 continue 

772 _flush_run() 

773 _flush_run() 

774 

775 plain, events = _detect_plain_numbered_toc(records, min_lines) 

776 toc.update(plain) 

777 return toc, events

780@dataclass(frozen=True) 

781class TocOutputPlan: 

782 """How a detected TOC is rendered into body output. 

783 

784 ``toc_indices`` stays the full heading-pipeline exclusion set; this plan 

785 only decides which of those excluded lines are nonetheless re-emitted as 

786 body, so a 目录 heading keeps a few of its entries instead of being 

787 orphaned. Budget is counted in VISIBLE LINES (soft-break lines count 

788 individually), matching DOCX_SMART_TOC_KEEP_LINES. 

789 """ 

790 

791 kept_text: dict[ 

792 int, str 

793 ] # record index -> body text (its first kept visible lines) 

794 ellipsis_anchor: ( 

795 int | None 

796 ) # emit one TOC_ELLIPSIS body line after this record; None = nothing elided 

797 kept_lines: int # visible lines retained as body 

798 removed_lines: int # visible lines elided (straddle tails + demoted_body_text) 

799 fully_removed_records: int # records with visible text and NOTHING kept 

800 

801 

802def plan_toc_output( 

803 records: Sequence[Any], toc_indices: set[int], *, keep_lines: int 

804) -> TocOutputPlan: 

805 """Decide which leading TOC lines to keep as body and where the single 

806 ``TOC_ELLIPSIS`` marker goes; count the line-level keep/drop. 

807 

808 Pure over ``(records, toc_indices, keep_lines)`` — the env read lives in 

809 the caller (``run_smart_heading``), so this stays trivially unit-testable 

810 with an explicit budget. Records absent from ``kept_text`` are dropped 

811 whole; a straddling record (one crossing the budget boundary) keeps its 

812 first lines and counts the remainder in ``removed_lines``. Invisible 

813 (whitespace-only) TOC records are skipped entirely — neither kept nor 

814 counted nor eligible as the ellipsis anchor. A record's 

815 ``demoted_body_text`` tail (oversize soft-break remainder from the read 

816 pass) is never re-emitted by the TOC branch, so its visible lines always 

817 count as removed — an elided tail alone still raises the ellipsis. 

818 

819 This is the single source for every disposition count the caller reports 

820 (``fully_removed_records`` included) — keep any new count here rather than 

821 re-deriving it at a call site, or the visibility definitions drift. 

822 """ 

823 ordered = [i for i in sorted(toc_indices) if records[i].kind == "para"] 

824 kept_text: dict[int, str] = {} 

825 remaining = max(0, keep_lines) 

826 kept_lines = 0 

827 removed_lines = 0 

828 fully_removed = 0 

829 last_kept: int | None = None 

830 first_visible: int | None = None 

831 for i in ordered: 

832 removed_lines += len(_visible_text_lines(records[i].demoted_body_text)) 

833 lines = _visible_lines(records[i]) 

834 if not lines: 

835 continue 

836 if first_visible is None: 

837 first_visible = i 

838 if remaining <= 0: 

839 removed_lines += len(lines) 

840 fully_removed += 1 

841 continue 

842 take = min(len(lines), remaining) 

843 kept_text[i] = "\n".join(lines[:take]) 

844 kept_lines += take 

845 removed_lines += len(lines) - take 

846 last_kept = i 

847 remaining -= take 

848 if removed_lines > 0: 

849 ellipsis_anchor = last_kept if last_kept is not None else first_visible 

850 else: 

851 ellipsis_anchor = None 

852 return TocOutputPlan( 

853 kept_text, ellipsis_anchor, kept_lines, removed_lines, fully_removed 

854 )

878def verify_content_preservation( 

879 records: Sequence[Any], 

880 blocks: Sequence[dict], 

881 *, 

882 toc_indices: set[int] = frozenset(), 

883 ignored_output_texts: Sequence[str] = (), 

884) -> list[str]: 

885 """I1 multiset check: every pass1-read paragraph's text must appear in 

886 the union of the output blocks' content∪heading. Returns the canonical 

887 forms that are MISSING (empty list = pass). 

888 

889 Exact line-level multiset matching first; leftovers get one merge 

890 tolerance pass — every leftover OUTPUT line must be exactly the 

891 concatenation of consecutive leftover source paragraphs in document 

892 order, each consumed at most once (covers heading merges). 

893 One-directional substring containment is not accepted: it would let a 

894 genuinely lost short paragraph hide inside an unrelated longer line. 

895 Only the TOC whitelist (indices from the three-channel judgment) may be 

896 absent entirely. 

897 

898 ``ignored_output_texts`` are output lines this call injected that do NOT 

899 come from any record (retained TOC lines + the ``TOC_ELLIPSIS`` marker); 

900 they are subtracted from the output multiset before matching so an injected 

901 copy cannot mask a genuinely lost source line (a leaderless TOC entry can 

902 canonicalize identically to its body heading). 

903 

904 Source paragraphs are split on ``\n`` before canonicalizing so a body 

905 paragraph carrying soft breaks (``w:br`` → ``"\n"``, emitted verbatim as 

906 multiple output content lines) compares symmetrically with the output; 

907 without the split the whole paragraph would canonicalize to one piece 

908 that no single output line equals, forcing a spurious fallback. Heading 

909 merges / soft-break heading joins still resolve through the tolerance 

910 pass (several source lines → one joined output line). 

911 """ 

912 source: list[str] = [] 

913 for i, rec in enumerate(records): 

914 if rec.kind != "para" or i in toc_indices: 

915 continue 

916 for piece in (rec.text, rec.demoted_body_text): 

917 if not piece: 

918 continue 

919 for line in piece.split("\n"): 

920 canon = canonicalize_paragraph_text(line) 

921 if canon: 921 ↛ 919line 921 didn't jump to line 919 because the condition on line 921 was always true

922 source.append(canon) 

923 

924 out_lines: dict[str, int] = {} 

925 for block in blocks: 

926 for line in (block.get("content") or "").split("\n"): 

927 canon = canonicalize_paragraph_text(line) 

928 if canon: 928 ↛ 926line 928 didn't jump to line 926 because the condition on line 928 was always true

929 out_lines[canon] = out_lines.get(canon, 0) + 1 

930 

931 # Subtract the lines this call injected (retained TOC body + the ellipsis) 

932 # at the same \n-split granularity as out_lines, so an injected copy cannot 

933 # satisfy a source line the assembler actually lost. 

934 for text in ignored_output_texts: 

935 for line in (text or "").split("\n"): 

936 canon = canonicalize_paragraph_text(line) 

937 if canon and out_lines.get(canon, 0) > 0: 937 ↛ 935line 937 didn't jump to line 935 because the condition on line 937 was always true

938 out_lines[canon] -= 1 

939 

940 missing: list[str] = [] 

941 leftovers_src: list[str] = [] 

942 for canon in source: 

943 count = out_lines.get(canon, 0) 

944 if count > 0: 

945 out_lines[canon] = count - 1 

946 else: 

947 leftovers_src.append(canon) 

948 

949 if leftovers_src: 

950 leftover_out: list[str] = [] 

951 for line, count in out_lines.items(): 

952 leftover_out.extend([line] * count) 

953 used = [False] * len(leftovers_src) 

954 for line in leftover_out: 

955 for s in range(len(leftovers_src)): 955 ↛ 954line 955 didn't jump to line 954 because the loop on line 955 didn't complete

956 if used[s] or not line.startswith(leftovers_src[s]): 

957 continue 

958 acc_len = 0 

959 picked: list[int] = [] 

960 k = s 

961 while k < len(leftovers_src) and acc_len < len(line): 

962 piece = leftovers_src[k] 

963 if used[k] or not line.startswith(piece, acc_len): 963 ↛ 964line 963 didn't jump to line 964 because the condition on line 963 was never true

964 break 

965 picked.append(k) 

966 acc_len += len(piece) 

967 k += 1 

968 if acc_len == len(line): 968 ↛ 955line 968 didn't jump to line 955 because the condition on line 968 was always true

969 for p in picked: 

970 used[p] = True 

971 break 

972 missing.extend( 

973 leftovers_src[s] for s in range(len(leftovers_src)) if not used[s] 

974 ) 

975 return missing

978def verify_baseline_heading_retention( 

979 records: Sequence[Any], 

980 decisions: Sequence[Any], 

981 *, 

982 demotion_rules: frozenset[str] = frozenset( 

983 { 

984 "strong_body_demoted", 

985 "clamp_gt9_demoted", 

986 "placeholder_demoted", 

987 "toc_removed", 

988 "title_block_member", 

989 "merged_absorbed", 

990 } 

991 ), 

992) -> list[int]: 

993 """I2: the smart heading set must cover every baseline (outlineLvl) 

994 heading except explicit rule-tagged demotions. Returns violating record 

995 indices (empty = pass). Table-cell paragraphs never become records, so 

996 that exemption is structural. 

997 """ 

998 by_record = {d.record_index: d for d in decisions} 

999 violations: list[int] = [] 

1000 for i, rec in enumerate(records):
