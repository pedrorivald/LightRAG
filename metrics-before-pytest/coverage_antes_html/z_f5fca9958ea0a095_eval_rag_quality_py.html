301 async def generate_rag_response( 

302 self, 

303 question: str, 

304 client: httpx.AsyncClient, 

305 ) -> Dict[str, Any]: 

306 """ 

307 Generate RAG response by calling LightRAG API. 

308 

309 Args: 

310 question: The user query. 

311 client: Shared httpx AsyncClient for connection pooling. 

312 

313 Returns: 

314 Dictionary with 'answer' and 'contexts' keys. 

315 'contexts' is a list of strings (one per retrieved document). 

316 

317 Raises: 

318 Exception: If LightRAG API is unavailable. 

319 """ 

320 try: 

321 payload = { 

322 "query": question, 

323 "mode": "mix", 

324 "include_references": True, 

325 "include_chunk_content": True, # NEW: Request chunk content in references 

326 "response_type": "Multiple Paragraphs", 

327 "top_k": int(os.getenv("EVAL_QUERY_TOP_K", "10")), 

328 } 

329 

330 # Get API key from environment for authentication 

331 api_key = os.getenv("LIGHTRAG_API_KEY") 

332 

333 # Prepare headers with optional authentication 

334 headers = {} 

335 if api_key: 

336 headers["X-API-Key"] = api_key 

337 

338 # Single optimized API call - gets both answer AND chunk content 

339 response = await client.post( 

340 f"{self.rag_api_url}/query", 

341 json=payload, 

342 headers=headers if headers else None, 

343 ) 

344 response.raise_for_status() 

345 result = response.json() 

346 

347 answer = result.get("response", "No response generated") 

348 references = result.get("references", []) 

349 

350 # DEBUG: Inspect the API response 

351 logger.debug("🔍 References Count: %s", len(references)) 

352 if references: 

353 first_ref = references[0] 

354 logger.debug("🔍 First Reference Keys: %s", list(first_ref.keys())) 

355 if "content" in first_ref: 

356 content_preview = first_ref["content"] 

357 if isinstance(content_preview, list) and content_preview: 

358 logger.debug( 

359 "🔍 Content Preview (first chunk): %s...", 

360 content_preview[0][:100], 

361 ) 

362 elif isinstance(content_preview, str): 

363 logger.debug("🔍 Content Preview: %s...", content_preview[:100]) 

364 

365 # Extract chunk content from enriched references 

366 # Note: content is now a list of chunks per reference (one file may have multiple chunks) 

367 contexts = [] 

368 for ref in references: 

369 content = ref.get("content", []) 

370 if isinstance(content, list): 

371 # Flatten the list: each chunk becomes a separate context 

372 contexts.extend(content) 

373 elif isinstance(content, str): 

374 # Backward compatibility: if content is still a string (shouldn't happen) 

375 contexts.append(content) 

376 

377 return { 

378 "answer": answer, 

379 "contexts": contexts, # List of strings from actual retrieved chunks 

380 } 

381 

382 except httpx.ConnectError as e: 

383 raise Exception( 

384 f"❌ Cannot connect to LightRAG API at {self.rag_api_url}\n" 

385 f" Make sure LightRAG server is running:\n" 

386 f" python -m lightrag.api.lightrag_server\n" 

387 f" Error: {str(e)}" 

388 ) 

389 except httpx.HTTPStatusError as e: 

390 raise Exception( 

391 f"LightRAG API error {e.response.status_code}: {e.response.text}" 

392 ) 

393 except httpx.ReadTimeout as e: 

394 raise Exception( 

395 f"Request timeout after waiting for response\n" 

396 f" Question: {question[:100]}...\n" 

397 f" Error: {str(e)}" 

398 ) 

399 except Exception as e: 

400 raise Exception(f"Error calling LightRAG API: {type(e).__name__}: {str(e)}")

402 async def evaluate_single_case( 

403 self, 

404 idx: int, 

405 test_case: Dict[str, str], 

406 rag_semaphore: asyncio.Semaphore, 

407 eval_semaphore: asyncio.Semaphore, 

408 client: httpx.AsyncClient, 

409 progress_counter: Dict[str, int], 

410 position_pool: asyncio.Queue, 

411 pbar_creation_lock: asyncio.Lock, 

412 ) -> Dict[str, Any]: 

413 """ 

414 Evaluate a single test case with two-stage pipeline concurrency control 

415 

416 Args: 

417 idx: Test case index (1-based) 

418 test_case: Test case dictionary with question and ground_truth 

419 rag_semaphore: Semaphore to control overall concurrency (covers entire function) 

420 eval_semaphore: Semaphore to control RAGAS evaluation concurrency (Stage 2) 

421 client: Shared httpx AsyncClient for connection pooling 

422 progress_counter: Shared dictionary for progress tracking 

423 position_pool: Queue of available tqdm position indices 

424 pbar_creation_lock: Lock to serialize tqdm creation and prevent race conditions 

425 

426 Returns: 

427 Evaluation result dictionary 

428 """ 

429 # rag_semaphore controls the entire evaluation process to prevent 

430 # all RAG responses from being generated at once when eval is slow 

431 async with rag_semaphore: 

432 question = test_case["question"] 

433 ground_truth = test_case["ground_truth"] 

434 

435 # Stage 1: Generate RAG response 

436 try: 

437 rag_response = await self.generate_rag_response( 

438 question=question, client=client 

439 ) 

440 except Exception as e: 

441 logger.error("Error generating response for test %s: %s", idx, str(e)) 

442 progress_counter["completed"] += 1 

443 return { 

444 "test_number": idx, 

445 "question": question, 

446 "error": str(e), 

447 "metrics": {}, 

448 "ragas_score": 0, 

449 "timestamp": datetime.now().isoformat(), 

450 } 

451 

452 # *** CRITICAL FIX: Use actual retrieved contexts, NOT ground_truth *** 

453 retrieved_contexts = rag_response["contexts"] 

454 

455 # Prepare dataset for RAGAS evaluation with CORRECT contexts 

456 eval_dataset = Dataset.from_dict( 

457 { 

458 "question": [question], 

459 "answer": [rag_response["answer"]], 

460 "contexts": [retrieved_contexts], 

461 "ground_truth": [ground_truth], 

462 } 

463 ) 

464 

465 # Stage 2: Run RAGAS evaluation (controlled by eval_semaphore) 

466 # IMPORTANT: Create fresh metric instances for each evaluation to avoid 

467 # concurrent state conflicts when multiple tasks run in parallel 

468 async with eval_semaphore: 

469 pbar = None 

470 position = None 

471 try: 

472 # Acquire a position from the pool for this tqdm progress bar 

473 position = await position_pool.get() 

474 

475 # Serialize tqdm creation to prevent race conditions 

476 # Multiple tasks creating tqdm simultaneously can cause display conflicts 

477 async with pbar_creation_lock: 

478 # Create tqdm progress bar with assigned position to avoid overlapping 

479 # leave=False ensures the progress bar is cleared after completion, 

480 # preventing accumulation of completed bars and allowing position reuse 

481 pbar = tqdm( 

482 total=4, 

483 desc=f"Eval-{idx:02d}", 

484 position=position, 

485 leave=False, 

486 ) 

487 # Give tqdm time to initialize and claim its screen position 

488 await asyncio.sleep(0.05) 

489 

490 eval_results = evaluate( 

491 dataset=eval_dataset, 

492 metrics=[ 

493 Faithfulness(), 

494 AnswerRelevancy(), 

495 ContextRecall(), 

496 ContextPrecision(), 

497 ], 

498 llm=self.eval_llm, 

499 embeddings=self.eval_embeddings, 

500 _pbar=pbar, 

501 ) 

502 

503 # Convert to DataFrame (RAGAS v0.3+ API) 

504 df = eval_results.to_pandas() 

505 

506 # Extract scores from first row 

507 scores_row = df.iloc[0] 

508 

509 # Extract scores (RAGAS v0.3+ uses .to_pandas()) 

510 result = { 

511 "test_number": idx, 

512 "question": question, 

513 "answer": rag_response["answer"][:200] + "..." 

514 if len(rag_response["answer"]) > 200 

515 else rag_response["answer"], 

516 "ground_truth": ground_truth[:200] + "..." 

517 if len(ground_truth) > 200 

518 else ground_truth, 

519 "project": test_case.get("project", "unknown"), 

520 "metrics": { 

521 "faithfulness": float(scores_row.get("faithfulness", 0)), 

522 "answer_relevance": float( 

523 scores_row.get("answer_relevancy", 0) 

524 ), 

525 "context_recall": float( 

526 scores_row.get("context_recall", 0) 

527 ), 

528 "context_precision": float( 

529 scores_row.get("context_precision", 0) 

530 ), 

531 }, 

532 "timestamp": datetime.now().isoformat(), 

533 } 

534 

535 # Calculate RAGAS score (average of all metrics, excluding NaN values) 

536 metrics = result["metrics"] 

537 valid_metrics = [v for v in metrics.values() if not _is_nan(v)] 

538 if not valid_metrics: 

539 # Nothing was scored (e.g. the judge LLM's output could 

540 # not be parsed for any metric). Report a failed case 

541 # rather than a 0.0 "success" that drags the averages 

542 # down as if the RAG answer itself were bad. 

543 raise ValueError( 

544 "All RAGAS metrics returned NaN; no score could be computed" 

545 ) 

546 ragas_score = sum(valid_metrics) / len(valid_metrics) 

547 result["ragas_score"] = round(ragas_score, 4) 

548 

549 # Update progress counter 

550 progress_counter["completed"] += 1 

551 

552 return result 

553 

554 except Exception as e: 

555 logger.error("Error evaluating test %s: %s", idx, str(e)) 

556 progress_counter["completed"] += 1 

557 return { 

558 "test_number": idx, 

559 "question": question, 

560 "error": str(e), 

561 "metrics": {}, 

562 "ragas_score": 0, 

563 "timestamp": datetime.now().isoformat(), 

564 } 

565 finally: 

566 # Force close progress bar to ensure completion 

567 if pbar is not None: 567 ↛ 570line 567 didn't jump to line 570 because the condition on line 567 was always true

568 pbar.close() 

569 # Release the position back to the pool for reuse 

570 if position is not None: 

571 await position_pool.put(position)

573 async def evaluate_responses(self) -> List[Dict[str, Any]]: 

574 """ 

575 Evaluate all test cases in parallel with two-stage pipeline and return metrics 

576 

577 Returns: 

578 List of evaluation results with metrics 

579 """ 

580 # Get evaluation concurrency from environment (default to 2 for parallel evaluation) 

581 max_async = int(os.getenv("EVAL_MAX_CONCURRENT", "2")) 

582 

583 logger.info("%s", "=" * 70) 

584 logger.info("🚀 Starting RAGAS Evaluation of LightRAG System") 

585 logger.info("🔧 RAGAS Evaluation (Stage 2): %s concurrent", max_async) 

586 logger.info("%s", "=" * 70) 

587 

588 # Create two-stage pipeline semaphores 

589 # Stage 1: RAG generation - allow x2 concurrency to keep evaluation fed 

590 rag_semaphore = asyncio.Semaphore(max_async * 2) 

591 # Stage 2: RAGAS evaluation - primary bottleneck 

592 eval_semaphore = asyncio.Semaphore(max_async) 

593 

594 # Create progress counter (shared across all tasks) 

595 progress_counter = {"completed": 0} 

596 

597 # Create position pool for tqdm progress bars 

598 # Positions range from 0 to max_async-1, ensuring no overlapping displays 

599 position_pool = asyncio.Queue() 

600 for i in range(max_async): 

601 await position_pool.put(i) 

602 

603 # Create lock to serialize tqdm creation and prevent race conditions 

604 # This ensures progress bars are created one at a time, avoiding display conflicts 

605 pbar_creation_lock = asyncio.Lock() 

606 

607 # Create shared HTTP client with connection pooling and proper timeouts 

608 # Timeout: 3 minutes for connect, 5 minutes for read (LLM can be slow) 

609 timeout = httpx.Timeout( 

610 TOTAL_TIMEOUT_SECONDS, 

611 connect=CONNECT_TIMEOUT_SECONDS, 

612 read=READ_TIMEOUT_SECONDS, 

613 )
