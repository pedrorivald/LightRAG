1"""Raw request-body ceiling as pure ASGI. 

2 

3``MAX_REQUEST_BODY_BYTES`` used to be enforced inside the admission middleware 

4and therefore covered exactly the three ingestion routes admission cares about. 

5That made the name a lie: an operator could set it, watch ``/documents/text`` 

6reject an 8 MiB body in milliseconds, and still have ``/api/chat`` — which needs 

7no credentials under the shipped whitelist — accept the same body and stall the 

8whole process tokenizing it (GHSA-r8jh-295g-vv42). 

9 

10The limit now lives here, applies to every route, and is layered because routes 

11legitimately differ by orders of magnitude in how much body they should accept: 

12 

13* ``/documents/upload`` derives its ceiling from ``MAX_UPLOAD_SIZE``, plus slack 

14 for multipart framing; 

15* ``/documents/text`` and ``/documents/texts`` get a generous ingestion ceiling, 

16 because a pasted document or a batch insert is not a chat turn and a batch has 

17 no equivalent on the upload route; 

18* everything else gets the default, which is what actually bounds the 

19 unauthenticated surface. 

20 

21Placement matters. This middleware is installed OUTSIDE the admission middleware 

22so an oversized body is refused before it can consume a capacity slot, and INSIDE 

23CORS so a 413 carries the headers a browser needs in order to read the status. 

24""" 

25 

26from __future__ import annotations 

27 

28from typing import Any, Optional 

29 

30from lightrag.utils import logger 

31 

32from .asgi_helpers import BodyLimitExceeded, header_value, limited_receive, send_json 

33from .utils_api import get_route_path

35# Routes whose legitimate body size differs from the default by orders of 

36# magnitude. Everything not listed here uses ``default_limit``. 

37UPLOAD_PATH = "/documents/upload" 

38INGEST_TEXT_PATHS: tuple[str, ...] = ("/documents/text", "/documents/texts") 

39 

40 

41class BodyLimitMiddleware: 

42 """Cut off a request body that exceeds the ceiling for its route. 

43 

44 Args: 

45 app: the wrapped ASGI application. 

46 default_limit: ceiling for ordinary routes; ``0`` disables. 

47 ingest_limit: ceiling for the text-ingestion routes; ``0`` disables. 

48 upload_limit: ceiling for ``/documents/upload``; ``0`` disables. 

49 api_prefix: mount prefix to strip before matching paths. 

50 """ 

51 

52 def __init__( 

53 self, 

54 app, 

55 *, 

56 default_limit: int = 0, 

57 ingest_limit: int = 0, 

58 upload_limit: int = 0, 

59 api_prefix: str = "", 

60 ) -> None: 

61 self.app = app 

62 self._default_limit = max(int(default_limit), 0) 

63 self._ingest_limit = max(int(ingest_limit), 0) 

64 self._upload_limit = max(int(upload_limit), 0) 

65 self._api_prefix = (api_prefix or "").rstrip("/") 

66 

67 def _limit_for(self, scope: dict[str, Any]) -> int: 

68 route = get_route_path(scope, self._api_prefix) 

69 if route == UPLOAD_PATH: 

70 return self._upload_limit 

71 if route in INGEST_TEXT_PATHS: 

72 return self._ingest_limit 

73 return self._default_limit 

74 

75 async def __call__(self, scope, receive, send) -> None: 

76 if scope["type"] != "http": 

77 await self.app(scope, receive, send) 

78 return 

79 

80 # Applied to every method rather than POST only: a body on GET or DELETE 

81 # is unusual but legal, and "unusual" is exactly where an unbounded read 

82 # would otherwise survive. Methods that carry no body make this a no-op. 

83 limit = self._limit_for(scope) 

84 if limit <= 0: 

85 await self.app(scope, receive, send) 

86 return 

87 

88 # An honest client's Content-Length lets the refusal happen without 

89 # touching receive() at all, so the body never leaves the client and 

90 # uvicorn never sends the "100 Continue" that would invite it. The 

91 # counting wrapper below is the actual enforcement. 

92 declared = header_value(scope, b"content-length") 

93 if declared and declared.isdigit() and int(declared) > limit: 

94 await send_json(send, 413, self._detail(limit)) 

95 return 

96 

97 response_started = False 

98 

99 async def _tracking_send(message: dict[str, Any]) -> None: 

100 nonlocal response_started 

101 if message.get("type") == "http.response.start": 

102 response_started = True 

103 await send(message) 

104 

105 try: 

106 await self.app(scope, limited_receive(receive, limit), _tracking_send) 

107 except BodyLimitExceeded as exceeded: 

108 logger.warning( 

109 f"Rejected {scope.get('path')}: request body exceeded the " 

110 f"{limit}-byte ceiling for this route after " 

111 f"{exceeded.args[0] if exceeded.args else '?'} bytes" 

112 ) 

113 if not response_started: 113 ↛ exitline 113 didn't return from function '__call__' because the condition on line 113 was always true

114 await send_json(send, 413, self._detail(limit)) 

115 # Otherwise the app already committed a status line; the connection 

116 # ends with a short body, which is all the protocol allows at that 

117 # point. Nothing is swallowed silently — the warning above is the 

118 # record. 

119 

120 @staticmethod 

121 def _detail(limit: int) -> str: 

122 return f"Request body too large. This endpoint accepts at most {limit} bytes."

125def resolve_body_limits(args: Any) -> Optional[dict[str, int]]: 

126 """Derive the three per-route ceilings from configuration. 

127 

128 Returns ``None`` when every ceiling is disabled, so the caller can skip 

129 installing the middleware entirely. 

130 

131 Semantics: 

132 

133 * ``MAX_REQUEST_BODY_BYTES = 0`` — an explicit opt-out. Every ceiling is 

134 off, including the derived upload one. 

135 * ``MAX_REQUEST_BODY_BYTES`` not configured — ordinary routes get the 

136 default, the text-ingestion routes get the built-in ingestion ceiling. 

137 * ``MAX_REQUEST_BODY_BYTES`` configured to any positive value — the 

138 operator's intent wins for every non-upload route, ingestion included. 

139 * ``MAX_UPLOAD_SIZE`` of ``0``/``None`` means "unlimited upload", a 

140 documented setting; there is then no size to derive a ceiling from, so the 

141 upload route is left uncapped. 

142 

143 Whether a value was configured is read from 

144 ``args.max_request_body_bytes_explicit`` and is deliberately NOT inferred by 

145 comparing the value to the default: an operator who sets the default value 

146 on purpose would be indistinguishable from one who set nothing, and the 

147 text-ingestion routes would silently get the 50 MiB tier instead of the 

148 ceiling that was asked for — widening a configured security boundary. 

149 

150 A caller that builds ``args`` by hand and omits the flag is treated as having 

151 configured the value. Guessing wrong in that direction costs a 413 on a large 

152 ingest, which is visible and recoverable; guessing wrong the other way 

153 silently relaxes a ceiling, which is the failure this parameter exists to 

154 prevent. 

155 """ 

156 from lightrag.constants import ( 

157 DEFAULT_MAX_INGEST_BODY_BYTES, 

158 MULTIPART_OVERHEAD_BYTES, 

159 ) 

160 

161 default_limit = max(int(getattr(args, "max_request_body_bytes", 0) or 0), 0) 

162 if default_limit <= 0: 

163 return None 

164 

165 configured = bool(getattr(args, "max_request_body_bytes_explicit", True)) 

166 ingest_limit = default_limit if configured else DEFAULT_MAX_INGEST_BODY_BYTES 

167 

168 max_upload_size = getattr(args, "max_upload_size", None) 

169 if isinstance(max_upload_size, int) and max_upload_size > 0: 

170 upload_limit = max_upload_size + MULTIPART_OVERHEAD_BYTES 

171 else: 

172 upload_limit = 0 

173 

174 return { 

175 "default_limit": default_limit, 

176 "ingest_limit": ingest_limit, 

177 "upload_limit": upload_limit, 

178 }
