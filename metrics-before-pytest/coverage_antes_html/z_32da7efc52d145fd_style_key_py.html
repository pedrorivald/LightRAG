1"""Numbering styleKey classification and FS_base statistics. 

2 

3Twelve numbering shapes, each with a unique ``styleKey``, a priority order 

4(smaller = shallower level when font sizes tie), optional unit-word sub-order 

5(篇/部/编/卷 > 章 > 节; 条 > 款 > 项; volume/part > chapter > section), and an 

6"allow empty title" flag. Classification is pure text analysis — candidacy 

7(font size / outline / bold gates) and homophone defenses (NER, blacklists) 

8live in the algorithm layer. 

9 

10Matching order: ``MultiLevelNum`` first (so ``1.1.2`` is not clipped by the 

11single-number rule), then the table order top-down, first hit wins 

12(``Section 3`` stops at EnChapter and never tries EnClause). Single-char 

13roman numerals (``I.`` / ``v.``) default to EnAlpha and are reclassified to 

14RomanNum by a deferred second scan only when a multi-char/Unicode RomanNum 

15companion exists in the same sub-document. 

16"""

60# --- the twelve patterns ----------------------------------------------------- 

61# Group 1 always captures the numbering prefix (label); the remainder of the 

62# paragraph after the full match is the title text. 

63 

64_P_MULTI_LEVEL = re.compile( 

65 # Separator "." must not be followed by a digit — otherwise a bare 

66 # trailing component ("1.2.3" as a whole paragraph) would backtrack into 

67 # "1.2" + separator "." + title "3" and defeat the no-empty-title rule. 

68 r"^\s*((?:§+\s*)?\d+(?:\.\d+)+)(?:[、\s]|\.(?!\d)|(?=[一-龥]))"

70# Shape probe: any multi-level number opener claims the paragraph for 

71# MultiLevelNum exclusively — when the full rule then rejects it (bare "1.2", 

72# "3.14"), the paragraph is body, and single-number rules (EnNum "1." + 

73# title "2") must NOT re-capture it.

95# Bounded repeated-letter group: Word list labels repeat the SAME letter 

96# (a, z, aa, zz, aaa). Mixed runs like "CV" or "MD" are abbreviations and 

97# must not classify as EnAlpha. 

98_P_EN_ALPHA = re.compile(r"^\s*(([A-Za-z])\2{0,2})[.、]") 

99_P_EN_DOUBLE_PAREN = re.compile(r"^\s*([（(](?:\d+|([A-Za-z])\2{0,2})[）)])") 

100_P_EN_SINGLE_PAREN = re.compile(r"^\s*((?:\d+|([A-Za-z])\2{0,2}))[）)]") 

101# Provenance-gated: mixed IVX runs (iv, vii, ix) are valid Roman list labels. 

102# The alphabetic backref above deliberately rejects those as non-Word alpha 

103# (CV. / MD. abbreviations); only lowerRoman/upperRoman numFmt may use these. 

104# Case-homogeneous like _P_ROMAN — a real label renders in one case. The \d+ 

105# branch is kept: a Roman numFmt still renders decimal when the resolver's 

106# _to_roman is out of domain (count <= 0 or >= 4000), and that label carries 

107# the Roman provenance, so dropping the branch would unclassify it. 

108_P_EN_DOUBLE_PAREN_ROMAN = re.compile(r"^\s*([（(](?:\d+|[IVX]+|[ivx]+|[Ⅰ-Ⅻⅰ-ⅻ])[）)])") 

109_P_EN_SINGLE_PAREN_ROMAN = re.compile(r"^\s*((?:\d+|[IVX]+|[ivx]+|[Ⅰ-Ⅻⅰ-ⅻ]))[）)]")

193def parse_roman(text: str) -> int | None: 

194 """Strict-ish subtractive roman parse over I/V/X; invalid forms → None.""" 

195 text = text.strip() 

196 if not text: 196 ↛ 197line 196 didn't jump to line 197 because the condition on line 196 was never true

197 return None 

198 if len(text) == 1 and text in _UNICODE_ROMAN_BASE: 

199 return _UNICODE_ROMAN_BASE[text] 

200 upper = text.upper() 

201 if any(ch not in _ROMAN_VALUES for ch in upper): 

202 return None 

203 total = 0 

204 prev = 0 

205 for ch in reversed(upper): 

206 val = _ROMAN_VALUES[ch] 

207 if val < prev: 

208 total -= val 

209 else: 

210 total += val 

211 prev = val 

212 # Round-trip check rejects malformed sequences like IIX. 

213 if _to_roman(total) != upper: 

214 return None

218def _to_roman(num: int) -> str | None: 

219 if not 0 < num < 40: # I..XXXIX covers the [IVX] alphabet 

220 return None 

221 out = [] 

222 for value, sym in ((10, "X"), (9, "IX"), (5, "V"), (4, "IV"), (1, "I")): 

223 while num >= value: 

224 out.append(sym) 

225 num -= value 

226 return "".join(out)

229def parse_alpha_ordinal(text: str) -> int | None: 

230 """Parse a homogeneous ASCII-letter run like ``a``, ``z``, ``aa``, ``zz``. 

231 

232 Word repeated-letter labels run the letter through the alphabet and count 

233 the run length as the round: ``a``/``z`` are 1/26, ``aa``/``zz`` are 

234 27/52, ``aaa`` is 53. Mixed runs (``ab``) and runs longer than 3 letters 

235 are not list labels and return None. 

236 """ 

237 text = text.strip() 

238 if not (text.isalpha() and text.isascii()): 

239 return None 

240 if not 1 <= len(text) <= 3: 

241 return None 

242 lowered = text.lower() 

243 if len(set(lowered)) != 1: 

244 return None 

245 letter_index = ord(lowered[0]) - ord("a") + 1 

246 return letter_index + 26 * (len(lowered) - 1)

352def _parse_marker_ordinal( 

353 text: str, *, numbering_format: str | None = None 

354) -> int | None: 

355 """Parse a bare list marker (no unit word), honoring DOCX provenance. 

356 

357 The parenthesized patterns accept repeated-letter runs, so an automatic 

358 lowerRoman/upperRoman list with lvlText "(%1)" hands "(ii)" here; read as 

359 alphabetic that is 35, not 2. _P_ROMAN cannot claim those labels (it only 

360 matches a "." or "、" terminator), so the carried numFmt is the only 

361 evidence available. When that provenance is Roman but ``parse_roman``

368 hand-typed markers on their existing behavior: "(i)" stays 9, disambiguated 

369 downstream by reclassify_single_char_romans rather than guessed here.

484@dataclass(frozen=True) 

485class FsBase: 

486 """Char-weighted dominant body font size and its confidence."""

492 

493def compute_fs_base( 

494 weighted_sizes: Iterable[tuple[float, int]], 

495 *, 

496 confidence_ratio: float = 0.60, 

497) -> FsBase: 

498 """Compute FS_base from ``(size_pt, char_weight)`` pairs. 

499 

500 The dominant size is the one covering the most characters; ties break 

501 toward the LARGER size. Confidence is "high" when the dominant size 

502 covers at least ``confidence_ratio`` of all weighted characters. 

503 """ 

504 weights: dict[float, int] = {} 

505 total = 0 

506 for size_pt, chars in weighted_sizes: 

507 if size_pt is None or chars <= 0: 507 ↛ 508line 507 didn't jump to line 508 because the condition on line 507 was never true

508 continue 

509 weights[size_pt] = weights.get(size_pt, 0) + chars 

510 total += chars 

511 if not weights or total <= 0: 

512 return FsBase(size_pt=None, dominant_ratio=0.0, confidence_high=False) 

513 size, weight = max(weights.items(), key=lambda kv: (kv[1], kv[0])) 

514 ratio = weight / total 

515 return FsBase( 

516 size_pt=size, 

517 dominant_ratio=ratio, 

518 confidence_high=ratio >= confidence_ratio, 

519 )
