6MIN_RAG_QUERY_WEIGHT = 3 

7 

8# East Asian characters count as two English-equivalent ones: a two-character 

9# CJK word carries about as much retrieval signal as a three-letter English one. 

10# 

11# WHOLE UNICODE BLOCKS, deliberately — never per assigned character, and never a 

12# boundary drawn inside a block. That choice is the point, not a shortcut: 

13# 

14# * It cannot rot. U+20000-U+3FFFD is the whole of planes 2 and 3, which 

15# Unicode allocates to CJK ideographs, so every extension from B through J 

16# and every future one is covered. A table enumerated per assigned character 

17# needs an edit — in two languages — on every Unicode release, and silently 

18# under-counts real characters until someone makes it. 

19# * The precision it gives up is worth nothing here. Punctuation, fillers and 

20# unassigned code points inside these ranges weigh 2, so two of them clear a 

21# minimum they should not. The cost of that is one retrieval that finds 

22# nothing. Nobody types "・・" meaning to ask something, and a 

23# three-character floor is a coarse heuristic for "did the user actually ask 

24# anything", not a correctness boundary. 

25# 

26# The blocks are the ones that write CHINESE, JAPANESE and KOREAN, which is what 

27# the minimum's contract says here, in both API guides and in all eleven locale 

28# strings. Scripts that are East Asian and wide but write other languages — 

29# Tangut, Khitan, Yi — are out. They were briefly in, from reading "East Asian 

30# wide" off a width table instead of the contract, and the only thing that came 

31# of it was a report that Tangut was covered inconsistently. Widening to every 

32# wide block would have answered that report by growing the table and invited 

33# the next one; matching the contract answers it by shrinking. 

34# 

35# Ranges therefore end on BLOCK boundaries. Ending one early is a real bug and 

36# has happened: carrying U+115F over from an older per-character table cut 

37# Hangul Jamo in half, so Korean medial and final jamo weighed 1. 

38# 

39# Keep these ranges in sync with ``lightrag_webui/src/utils/queryValidation.ts``; 

40# ``test_the_frontend_range_table_matches_this_one`` enforces it. 

41_WIDE_CODEPOINT_RANGES = ( 

42 (0x1100, 0x11FF), # Hangul Jamo 

43 (0x2E80, 0x33FF), # CJK Radicals, Kangxi, Symbols, Kana, Bopomofo, Hangul 

44 (0x3400, 0x4DBF), # CJK Unified Ideographs Extension A 

45 (0x4E00, 0x9FFF), # CJK Unified Ideographs 

46 (0xA960, 0xA97F), # Hangul Jamo Extended-A 

47 (0xAC00, 0xD7FF), # Hangul Syllables and Hangul Jamo Extended-B 

48 (0xF900, 0xFAFF), # CJK Compatibility Ideographs 

49 (0xFE30, 0xFE4F), # CJK Compatibility Forms 

50 (0xFF00, 0xFFEE), # Halfwidth and Fullwidth Forms 

51 (0x16FE0, 0x16FFF), # Ideographic Symbols and Punctuation 

52 (0x1AFF0, 0x1B2FF), # Kana Extended-B/Supplement/Extended-A, Small Kana, Nushu 

53 (0x20000, 0x3FFFD), # Planes 2 and 3 — every CJK extension, present and future 

54)

64class QueryValidationError(ValueError): 

65 """Base class for rejections of user-supplied query text."""

68class EmptyQueryError(QueryValidationError): 

69 """Raised when query text is empty or only whitespace. 

70 

71 This applies to every path, ``bypass`` and ``/api/generate`` included: an 

72 empty prompt has no meaning to send to an LLM, and forwarding it only buys 

73 a provider-side error or a hallucinated answer to nothing. 

74 """

77class RAGQueryTooShortError(QueryValidationError): 

78 """Raised when query text is too short for meaningful RAG retrieval."""

81def _is_wide_character(character: str) -> bool: 

82 codepoint = ord(character) 

83 return any(start <= codepoint <= end for start, end in _WIDE_CODEPOINT_RANGES)

86def rag_query_weight(query: str) -> int: 

87 """Return English-equivalent query length after trimming outer whitespace. 

88 

89 Each East Asian character counts as two; every other Unicode code point 

90 counts as one. Internal whitespace remains part of the query, matching the 

91 previous character-length contract. 

92 

93 Prefer :func:`meets_min_rag_query_weight` when only the comparison against 

94 :data:`MIN_RAG_QUERY_WEIGHT` matters: this function walks the whole string, 

95 which is up to 64 KiB of synchronous work on the event loop. 

96 """ 

97 

98 return sum(2 if _is_wide_character(char) else 1 for char in query.strip()) 

99 

100 

101def meets_min_rag_query_weight(query: str) -> bool: 

102 """Return whether ``query`` reaches :data:`MIN_RAG_QUERY_WEIGHT`. 

103 

104 Short-circuits as soon as the threshold is reached, so a 64 KiB query costs 

105 two iterations rather than 65,536. The full :func:`rag_query_weight` walk 

106 added ~69 ms of event-loop-blocking CPU to every max-size RAG request. 

107 """ 

108 

109 weight = 0 

110 for character in query.strip(): 

111 weight += 2 if _is_wide_character(character) else 1 

112 if weight >= MIN_RAG_QUERY_WEIGHT: 

113 return True 

114 return False

117def validate_query_not_empty(query: str) -> str: 

118 """Normalize query text and reject it when nothing but whitespace remains. 

119 

120 Applies to every mode, including ``bypass`` and ``/api/generate``, which are 

121 exempt from the RAG minimum but not from having to say something. 

122 """ 

123 

124 normalized = query.strip() 

125 if not normalized: 

126 raise EmptyQueryError("Query must not be empty.") 

127 return normalized 

128 

129 

130def validate_rag_query(query: str) -> str: 

131 """Normalize and validate query text used by RAG retrieval modes.""" 

132 

133 normalized = validate_query_not_empty(query) 

134 if not meets_min_rag_query_weight(normalized): 

135 raise RAGQueryTooShortError(_TOO_SHORT_MESSAGE) 

136 return normalized
