26class NativeDocxParser(NativeParserBase): 

27 """Native DOCX parser for LightRAG's production parsing path. 

28 

29 ``extract_docx_blocks`` performs only heading-driven structural splitting 

30 (one block per DOCX heading). Block sizing is intentionally left to the 

31 downstream paragraph-semantic chunker, so this parser emits the 

32 one-heading-one-block sidecar contract that chunking consumes. 

33 """ 

34 

35 engine_name = PARSER_ENGINE_NATIVE 

36 empty_content_label = "DOCX"

38 def validate_source(self, source: Path, file_path: str) -> None: 

39 if not ( 

40 source.exists() and source.is_file() and source.suffix.lower() == ".docx" 

41 ): 

42 raise ValueError( 

43 f"Native parser does not support pending file: {file_path}"

46 def validate_source_blocking(self, source: Path, file_path: str) -> None: 

47 # Precedes every reader of this archive: the base calls this at the 

48 # top of _extract_sync, and extract is what opens the package 

49 # (python-docx plus four more ZipFile opens). Nothing downstream 

50 # bounds the expansion. 

51 # 

52 # Deliberately NOT in validate_source: reading the central directory 

53 # of a .docx with hundreds of thousands of members takes seconds 

54 # (measured: 24s for 500k), and validate_source runs on the event 

55 # loop, where that would stall every unrelated request — the very 

56 # symptom this budget exists to prevent. 

57 enforce_docx_decompression_budget(source, file_path)

120 def finalize_parse_warnings( 

121 self, 

122 warnings: dict[str, Any], 

123 metadata: dict[str, Any], 

124 *, 

125 parsed_dir: Path, 

126 base_name: str, 

127 source: Path, 

128 i4_cache_disabled: bool, 

129 ) -> dict[str, Any] | None: 

130 """Divert smart-heading diagnostics to the sidecar; keep the rest. 

131 

132 Smart-heading warnings (``smart_*`` / ``title_block_*``) are merged 

133 under a ``parse_warnings`` key into the audit ledger docx left in 

134 ``metadata["smart_audit"]`` and written once to 

135 ``<base>.smart_audit.json`` (timestamp-free + ``sort_keys`` so a 

136 re-parse stays byte-identical — I4). Every other warning (missing 

137 paraId, over-long-heading handling) is returned unconditionally so the 

138 pipeline keeps mirroring it onto ``doc_status.metadata``. 

139 """ 

140 # I4 determinism waiver: a smart-heading LLM ran with the entity-extract 

141 # cache off. Record it in the audit file — the base no longer injects 

142 # this smart_ key into the warnings dict. 

143 if i4_cache_disabled: 

144 warnings["smart_i4_cache_disabled"] = 1 

145 

146 missing = int(warnings.get("missing_paraid_count", 0) or 0) 

147 if missing > 0: 

148 # Surface once per document; affected blocks emit 

149 # ``positions: [{"type": "paraid", "range": null}]``. 

150 logger.warning( 

151 "[parse_native] %s: %d paragraphs lack paraId; " 

152 "Re-saving file in Word 2013+ to regenerate ids.", 

153 source.name, 

154 missing, 

155 ) 

156 

157 smart = { 

158 k: v 

159 for k, v in warnings.items() 

160 if k.startswith(_SMART_HEADING_WARNING_PREFIXES) and v 

161 } 

162 other = { 

163 k: v 

164 for k, v in warnings.items() 

165 if not k.startswith(_SMART_HEADING_WARNING_PREFIXES) and v 

166 } 

167 

168 # Merge into the ledger (pop before build_ir sees it) and write the 

169 # sidecar. Written whenever there is any audit content — ledger only, 

170 # warnings only, or both. 

171 ledger = metadata.pop("smart_audit", None) 

172 audit = dict(ledger) if isinstance(ledger, dict) else {} 

173 if smart: 

174 audit["parse_warnings"] = smart 

175 if audit: 

176 audit_path = parsed_dir / f"{base_name}.smart_audit.json" 

177 audit_path.write_text( 

178 json.dumps(audit, ensure_ascii=False, indent=2, sort_keys=True) + "\n", 

179 encoding="utf-8", 

180 ) 

181 

182 return other or None
