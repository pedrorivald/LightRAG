71# Storage-managed ``create_time`` is serialized as the FIRST key of every KV 

72# value so it can be recovered with a bounded prefix read instead of pulling 

73# the whole row back over the wire -- a ``full_docs`` row is an entire 

74# document, and Redis has no way to read one JSON field. 64 bytes covers 

75# ``{"create_time":`` plus a 19-digit integer and its comma. 

76# 

77# The separator is matched loosely on purpose: ``json.dumps`` writes 

78# ``{"create_time": 1700000000,`` (a space after the colon) while a compact 

79# dump or a hand-edited row writes it without one. A prefix that does not 

80# match is not a failure -- it falls back to a full read -- but a needlessly 

81# strict pattern would silently send EVERY update down that fallback and 

82# quietly undo the optimization. 

83_CREATE_TIME_PREFIX_BYTES = 64 

84_CREATE_TIME_PREFIX_RE = re.compile(r'^\{\s*"create_time"\s*:\s*(-?\d+)\s*[,}]') 

85 

86# Lua counterpart of _CREATE_TIME_PREFIX_RE. Keep the two in step; the 

87# integration suite pins them to the same verdict on the same prefixes. 

88_CREATE_TIME_PREFIX_LUA_PATTERN = '^{%s*"create_time"%s*:%s*(%-?%d+)%s*[,}]'

90# The script that writes a KV row. Reading the previous ``create_time`` and 

91# writing the new value must be ONE atomic step, or a concurrent ``delete()`` 

92# landing between them lets the row come back carrying the deleted 

93# incarnation's timestamp -- and nothing would ever correct it, because every 

94# later update preserves what it finds. Redis runs a script atomically, so the 

95# decision happens next to the data: 

96# 

97# * key absent -> stamp ARGV[3] (the write's own clock). Two 

98# writers racing a first insert therefore cannot disagree: the second one 

99# to run finds the row and keeps the first one's timestamp. ``GETRANGE`` 

100# answers ``''`` for a missing key AND for a key holding an empty string, 

101# so an ``EXISTS`` disambiguates them -- only on that branch, so the 

102# steady state still reads once. An empty row is a stored row with no 

103# usable timestamp, and the contract says such a row records ``0``; it 

104# must never be mistaken for a first creation and given this clock. 

105# * prefix carries an int -> keep it. This is the steady state, and it needs 

106# nothing from the caller: one round trip, no read amplification. 

107# * prefix does not match -> a row written before this layout existed, or by 

108# hand. The timestamp may sit anywhere in the value or be malformed, and 

109# normalizing it belongs to ``normalize_kv_create_time`` and not to a 

110# second implementation in Lua -- so the script WRITES NOTHING and answers 

111# ``needs_hint``. The caller reads the value, normalizes it, and calls 

112# again with ARGV[2] set; a non-empty hint always writes, so the exchange 

113# cannot loop. 

114# 

115# No ``cjson`` and no arithmetic: the timestamp travels as a string and the 

116# value is spliced in front of the caller's payload, so a large row is never 

117# decoded server-side and Lua's number formatting never enters the picture. 

118_CREATE_TIME_UPSERT_LUA = f""" 

119local prefix = redis.call('GETRANGE', KEYS[1], 0, tonumber(ARGV[4]) - 1) 

120local outcome 

121local create_time 

122if prefix == '' and redis.call('EXISTS', KEYS[1]) == 0 then 

123 create_time = ARGV[3] 

124 outcome = 'created' 

125else 

126 local matched = string.match(prefix, '{_CREATE_TIME_PREFIX_LUA_PATTERN}') 

127 if matched then 

128 create_time = matched 

129 outcome = 'kept' 

130 elseif ARGV[2] == '' then 

131 return {{'needs_hint', ''}} 

132 else 

133 create_time = ARGV[2] 

134 outcome = 'hinted' 

135 end 

136end 

137local rest = string.sub(ARGV[1], 2) 

138if rest == '}}' or rest == '' then 

139 redis.call('SET', KEYS[1], '{{"create_time":' .. create_time .. '}}') 

140else 

141 redis.call('SET', KEYS[1], '{{"create_time":' .. create_time .. ',' .. rest) 

142end 

143return {{outcome, create_time}} 

144"""
