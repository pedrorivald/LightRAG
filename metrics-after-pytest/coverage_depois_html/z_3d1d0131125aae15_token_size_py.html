1"""Fixed-size token-window chunking — the LightRAG default strategy. 

2 

3Chunks the input text into windows of at most ``chunk_token_size`` tokens 

4with ``chunk_overlap_token_size`` of overlap between adjacent windows. 

5When ``split_by_character`` is supplied, the splitter first segments on 

6that delimiter and then either tokenizes each segment as-is 

7(``split_by_character_only=True``) or further sub-splits any segment 

8that exceeds the token cap. 

9 

10Two entry points are exported: 

11 

12 - :func:`chunking_by_token_size` — the **legacy 6-arg signature** 

13 used as the default value for :attr:`lightrag.LightRAG.chunking_func`. 

14 Kept for backward compatibility so externally-supplied chunking 

15 functions can continue to drop in unchanged. 

16 

17 - :func:`chunking_by_fixed_token` — the same algorithm exposed under 

18 the **new file-chunker contract** (standard prefix 

19 ``(tokenizer, content, chunk_token_size)`` plus keyword-only 

20 knobs). Used by the file-based chunking dispatcher in 

21 ``process_single_document`` for ``doc_process_opts.chunking == "F"``. 

22"""

32def _trimmed_span(content: str, start: int, end: int) -> tuple[int, int]: 

33 """Return the source span after applying the chunker's ``.strip()``.""" 

34 start = max(0, min(start, len(content))) 

35 end = max(start, min(end, len(content))) 

36 while start < end and content[start].isspace(): 

37 start += 1 

38 while end > start and content[end - 1].isspace(): 

39 end -= 1 

40 return start, end

43def _source_span(content: str, start: int, end: int) -> dict[str, int] | None: 

44 start, end = _trimmed_span(content, start, end) 

45 if start >= end: 

46 return None 

47 return {"start": start, "end": end}

50def _token_window_source_span( 

51 tokenizer: Tokenizer, 

52 content: str, 

53 tokens: list[int], 

54 start_token: int, 

55 end_token: int, 

56 *, 

57 anchor: tuple[int, int], 

58) -> tuple[dict[str, int] | None, tuple[int, int]]: 

59 """Map a decoded token window back to its exact source span. 

60 

61 ``anchor`` is the previous window's *verified* ``(start_token, start_char)``. 

62 Window starts are monotonically increasing, so instead of re-decoding the whole 

63 ``tokens[:start_token]`` prefix (O(N) per window → O(N²) overall) we decode only 

64 the delta ``tokens[anchor_token:start_token]`` (≈ one chunking step) to predict 

65 the start char. The predicted offset is then verified against ``content`` exactly 

66 as a full prefix decode would be: byte-level BPE decode is non-concatenative at a 

67 multi-byte UTF-8 boundary, so a delta can be off by the few chars of one split 

68 char — the ±32 ``find`` fallback corrects that, and re-anchoring on the verified 

69 position each call keeps the error from accumulating. Net cost is O(N) total 

70 while the located span stays byte-exact. 

71 

72 Returns ``(span, new_anchor)``. On an unlocatable (U+FFFD) window the span is 

73 ``None`` and the anchor is returned unchanged so the next window still predicts 

74 from the last verified position. 

75 """ 

76 anchor_token, anchor_char = anchor 

77 window = tokenizer.decode(tokens[start_token:end_token]) 

78 if start_token >= anchor_token: 78 ↛ 81line 78 didn't jump to line 81 because the condition on line 78 was always true

79 start = anchor_char + len(tokenizer.decode(tokens[anchor_token:start_token])) 

80 else: # non-monotonic caller (not expected) — fall back to a full prefix decode 

81 start = len(tokenizer.decode(tokens[:start_token])) 

82 end = start + len(window) 

83 if content[start:end] != window: 

84 found = content.find( 

85 window, 

86 max(0, start - 32), 

87 min(len(content), end + 32 + len(window)), 

88 ) 

89 if found < 0: 89 ↛ 91line 89 didn't jump to line 91 because the condition on line 89 was always true

90 return None, anchor 

91 start = found 

92 end = found + len(window) 

93 return _source_span(content, start, end), (start_token, start)

114def _window_step(chunk_token_size: int, chunk_overlap_token_size: int) -> int: 

115 """Token-window stride for the sliding-window chunk loops. 

116 

117 When overlap >= size the stride is <= 0, which makes ``range()`` yield an 

118 empty sequence (dropping the whole segment silently) or raise the opaque 

119 ``range() arg 3 must not be zero``. Fail closed with the same invariant the 

120 API-boundary validators enforce. 

121 

122 Negative overlap is rejected up front in ``chunking_by_token_size()``, 

123 before this function is ever reached -- see the check there. 

124 """ 

125 if chunk_overlap_token_size >= chunk_token_size: 

126 raise ValueError( 

127 f"chunk_overlap_token_size ({chunk_overlap_token_size}) must be < " 

128 f"chunk_token_size ({chunk_token_size})" 

129 ) 

130 return chunk_token_size - chunk_overlap_token_size

133def chunking_by_token_size( 

134 tokenizer: Tokenizer, 

135 content: str, 

136 split_by_character: str | None = None, 

137 split_by_character_only: bool = False, 

138 chunk_overlap_token_size: int = 100, 

139 chunk_token_size: int = 1200, 

140 *, 

141 _emit_source_span: bool = False, 

142) -> list[dict[str, Any]]: 

143 """Legacy 6-arg fixed-token chunker (default for ``LightRAG.chunking_func``). 

144 

145 Signature is preserved for backward compatibility with externally 

146 supplied ``chunking_func`` implementations. New file-based chunking 

147 dispatch uses :func:`chunking_by_fixed_token` instead. 

148 """ 

149 # Checked up front, before any branching on split_by_character, so a 

150 # negative chunk_overlap_token_size is always rejected -- including on 

151 # the split_by_character path where every segment stays under 

152 # chunk_token_size and _window_step()'s own (upper-bound only) check 

153 # would otherwise never be reached. The upper-bound check stays lazy, 

154 # inside _window_step(), since it's only meaningful once windowing is 

155 # actually about to happen. 

156 if chunk_overlap_token_size < 0: 

157 raise ValueError( 

158 f"chunk_overlap_token_size must be non-negative, got " 

159 f"{chunk_overlap_token_size}" 

160 ) 

161 results: list[dict[str, Any]] = [] 

162 if split_by_character: 

163 raw_chunks = content.split(split_by_character) 

164 raw_spans: list[tuple[int, int]] = [] 

165 cursor = 0 

166 for raw_chunk in raw_chunks: 

167 start = cursor 

168 end = start + len(raw_chunk) 

169 raw_spans.append((start, end)) 

170 cursor = end + len(split_by_character) 

171 new_chunks = [] 

172 if split_by_character_only: 

173 for chunk, (chunk_start, chunk_end) in zip(raw_chunks, raw_spans): 

174 _tokens = tokenizer.encode(chunk) 

175 if len(_tokens) > chunk_token_size: 

176 logger.warning( 

177 "Chunk split_by_character exceeds token limit: len=%d limit=%d", 

178 len(_tokens), 

179 chunk_token_size, 

180 ) 

181 raise ChunkTokenLimitExceededError( 

182 chunk_tokens=len(_tokens), 

183 chunk_token_limit=chunk_token_size, 

184 chunk_preview=chunk[:120], 

185 ) 

186 span = ( 

187 _source_span(content, chunk_start, chunk_end) 

188 if _emit_source_span 

189 else None 

190 ) 

191 new_chunks.append((len(_tokens), chunk, span)) 

192 else: 

193 for chunk, (chunk_start, chunk_end) in zip(raw_chunks, raw_spans): 

194 _tokens = tokenizer.encode(chunk) 

195 if len(_tokens) > chunk_token_size: 

196 # Anchor is chunk-relative (offsets are shifted by chunk_start 

197 # below), so it resets per split-by-character segment. 

198 anchor = (0, 0) 

199 for start in range( 

200 0, 

201 len(_tokens), 

202 _window_step(chunk_token_size, chunk_overlap_token_size), 

203 ): 

204 end_token = min(start + chunk_token_size, len(_tokens)) 

205 chunk_content = tokenizer.decode(_tokens[start:end_token]) 

206 span = None 

207 if _emit_source_span: 207 ↛ 208line 207 didn't jump to line 208 because the condition on line 207 was never true

208 span, anchor = _token_window_source_span( 

209 tokenizer, 

210 chunk, 

211 _tokens, 

212 start, 

213 end_token, 

214 anchor=anchor, 

215 ) 

216 if span is not None: 216 ↛ 217line 216 didn't jump to line 217 because the condition on line 216 was never true

217 span = { 

218 "start": chunk_start + span["start"], 

219 "end": chunk_start + span["end"], 

220 } 

221 new_chunks.append( 

222 ( 

223 min(chunk_token_size, len(_tokens) - start), 

224 chunk_content, 

225 span, 

226 ) 

227 ) 

228 else: 

229 span = ( 

230 _source_span(content, chunk_start, chunk_end) 

231 if _emit_source_span 

232 else None 

233 ) 

234 new_chunks.append((len(_tokens), chunk, span)) 

235 for index, (_len, chunk, span) in enumerate(new_chunks): 

236 results.append( 

237 _make_chunk( 

238 content=chunk, 

239 tokens=_len, 

240 order=index, 

241 source_span=span, 

242 emit_source_span=_emit_source_span, 

243 ) 

244 ) 

245 else: 

246 # Only the fixed-window path needs the whole-document token list; 

247 # the split_by_character path re-encodes each segment anyway (BPE 

248 # boundaries differ on substrings), so the full encode is deferred 

249 # here rather than paid on every call. 

250 tokens = tokenizer.encode(content) 

251 anchor = (0, 0) 

252 for index, start in enumerate( 

253 range( 

254 0, 

255 len(tokens), 

256 _window_step(chunk_token_size, chunk_overlap_token_size), 

257 ) 

258 ): 

259 end = min(start + chunk_token_size, len(tokens)) 

260 chunk_content = tokenizer.decode(tokens[start:end]) 

261 span = None 

262 if _emit_source_span: 

263 span, anchor = _token_window_source_span( 

264 tokenizer, content, tokens, start, end, anchor=anchor 

265 ) 

266 results.append( 

267 _make_chunk( 

268 content=chunk_content, 

269 tokens=min(chunk_token_size, len(tokens) - start), 

270 order=index, 

271 source_span=span, 

272 emit_source_span=_emit_source_span, 

273 ) 

274 ) 

275 return results
