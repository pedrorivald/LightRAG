1"""Title-block (document main title) discovery and LLM judgment. 

2 

3Heuristics find CANDIDATE windows — runs of paragraphs lacking strong-body 

4features that contain a visually dominant line — plus a single-paragraph 

5channel limited to the document's first eligible paragraph. An LLM (via the 

6synchronous judge callable; this module never touches asyncio) then confirms 

7and decomposes each candidate. 

8 

9LLM output is STRICTLY validated: the main/sub title must be locatable in 

10the window text, and paragraphs carrying a physical outline level are never 

11demoted by an LLM "body" vote (invariant I2). Locate-back rejects INVENTED 

12text, not editorial polish — beyond a literal substring it accepts a 

13concatenation of consecutive paragraphs, punctuation/width/case 

14normalization, and a few long window fragments reassembled in source order 

15(the overlap-dedup case where two cover lines share a boundary word); see 

16:func:`_locate`, whose relaxed hits are counted in the 

17``title_block_locate_relaxed`` parse warning. A 

18non-title verdict's headings/body partition must be well-formed; a MALFORMED 

19partition (missing/null field, out-of-range index, a duplicate within a list, 

20or the same index voted both heading and body) raises 

21:class:`TitleBlockLLMError`, as does any unparseable or non-locatable answer — 

22those failures are loud, never silently degraded. An UNDER-SPECIFIED partition 

23(two valid lists whose union merely omits some window indices) is instead 

24recovered: the unmentioned paragraphs abstain (named in neither list, so 

25neither audited as headings nor vetoed) and re-enter the normal flow, logged 

26via ``title_block_partition_incomplete`` so a single local omission never 

27fails the whole document. Of the two partition lists only the BODY side 

28carries force (a veto revoking candidate identity); the headings side is 

29retained for audit/reference and never admits a heading by itself. 

30"""

131def _fold(text: str) -> str: 

132 """Punctuation/width/case-insensitive form of an already-canon string. 

133 

134 NFKC unifies full-width and half-width forms (``ＡＢ`` → ``AB``, ``：`` → 

135 ``:``), every punctuation/symbol character is dropped so an added, 

136 removed, or swapped separator cannot break the match (``2027-2028`` vs 

137 ``2027—2028``, ``（2026年修订）`` vs ``(2026年修订)``), and case folds for 

138 Latin text. NFKC can re-introduce whitespace (``\\u00a0`` → space), so 

139 whitespace is stripped here too rather than relying on the caller. 

140 """ 

141 folded = [] 

142 for ch in unicodedata.normalize("NFKC", text or ""): 

143 if ch.isspace(): 143 ↛ 144line 143 didn't jump to line 144 because the condition on line 143 was never true

144 continue 

145 if unicodedata.category(ch)[0] in ("P", "S"): 

146 continue 

147 folded.append(ch) 

148 return "".join(folded).casefold()

151def _assemble_from_window(answer: str, window: str) -> bool: 

152 """True when ``answer`` is a concatenation of at most 

153 :data:`_LOCATE_MAX_PIECES` ``window`` fragments taken in source order. 

154 

155 Both arguments must already be :func:`_fold`-ed. Fragments may not 

156 overlap and may not go backwards (each search resumes after the previous 

157 fragment), so the answer's characters are the window's characters in the 

158 window's own order — the property the hallucination guard actually 

159 protects. Longest-fragment-first with backtracking, bounded by 

160 :data:`_LOCATE_ASSEMBLE_STEPS` probes; exhausting the budget reports 

161 "not located" (the strict, safe direction). 

162 """ 

163 budget = [_LOCATE_ASSEMBLE_STEPS] 

164 

165 def walk(start: int, from_pos: int, pieces_left: int) -> bool: 

166 if start == len(answer): 

167 return True 

168 if pieces_left == 0: 

169 return False 

170 # Longest first: the real split is one long fragment plus a short 

171 # remainder, and a long first fragment prunes the search hardest. 

172 for end in range(len(answer), start, -1): 172 ↛ 184line 172 didn't jump to line 184 because the loop on line 172 didn't complete

173 piece = answer[start:end] 

174 if guardrails.weighted_char_length(piece) < _LOCATE_MIN_PIECE_WEIGHTED: 

175 break # every shorter prefix is shorter still 

176 pos = window.find(piece, from_pos) 

177 while pos != -1: 

178 if budget[0] <= 0: 178 ↛ 179line 178 didn't jump to line 179 because the condition on line 178 was never true

179 return False 

180 budget[0] -= 1 

181 if walk(end, pos + len(piece), pieces_left - 1): 

182 return True 

183 pos = window.find(piece, pos + 1) 

184 return False 

185 

186 return walk(0, 0, _LOCATE_MAX_PIECES)

200def flatten_heading_line(text: str) -> str: 

201 """Render a heading that carries line breaks as ONE line. 

202 

203 A heading is a single title no matter how its source was line-wrapped 

204 (soft ``<w:br>`` breaks, an LLM echoing a multi-paragraph cover title) — 

205 left multi-line it leaks ``\\n`` into every descendant's 

206 ``parent_headings``. ``splitlines()`` covers ``\\r\\n``/U+2028-style 

207 breaks, not just ``\\n``; lines join CJK-aware so Chinese titles gain no 

208 stray spaces. 

209 """ 

210 lines = [ln.strip() for ln in text.splitlines()] 

211 lines = [ln for ln in lines if ln] 

212 if not lines: 

213 return "" 

214 joined = lines[0] 

215 for ln in lines[1:]: 

216 joined = _join_heading_texts(joined, ln) 

217 return joined

402def detect_imprint_regions( 

403 records: Sequence[Any], 

404 *, 

405 imprint_marker: Callable[[str], str | None], 

406 imprint_closer: Callable[[str], str | None], 

407 document_date: Callable[[str], bool] | None = None, 

408 skip_indices: set[int] = frozenset(), 

409) -> list[ImprintRegion]: 

410 """Map every 公文版记 region: an imprint anchor with its neighbourhood. 

411 

412 For each anchor (:func:`guardrails.imprint_marker_reason`, e.g. 抄送 / 

413 主题词; TOC lines cannot anchor) two bounded walks run: 

414 

415 - BACKWARD: up to 2 non-blank paragraphs → ``preceding`` (the signature / 

416 date lines above the 版记). 

417 - FORWARD: up to ``DOCX_SMART_IMPRINT_FORWARD_PARAS`` (default 3) non-blank 

418 paragraphs, closing on the first 印发-family CLOSER 

419 (:func:`guardrails.imprint_closer_reason`, only ever recognized here, in 

420 an anchor's forward window). Another anchor (抄送/主题词) encountered mid- 

421 walk is MIDDLE content, not a closer — a 主题词-opened region runs THROUGH 

422 a following 抄送 to reach the 印发 closer. When a closer is found the 

423 anchor→closer span (middle lines included) becomes ``members``; otherwise 

424 ``members`` is just the anchor. 

425 - After the closer, up to 2 bare 成文日期 lines (:func:`guardrails. 

426 is_document_date`) immediately following it are ALSO pulled into 

427 ``members``. Some documents mis-order 版记 THEN 成文日期 (not the GB/T 

428 order); that trailing date belongs to THIS document, so absorbing it 

429 keeps the NEXT cover from seeding a title window on it. 

430 

431 All walks step over blank AND TOC lines without spending budget, and stop 

432 at any non-paragraph record (a table / section break is a structural 

433 boundary a ruling must not leak across). Overlapping anchors' regions are 

434 unioned by the caller. 

435 """ 

436 document_date = document_date or guardrails.is_document_date 

437 forward_paras = _env_int( 

438 "DOCX_SMART_IMPRINT_FORWARD_PARAS", DEFAULT_DOCX_SMART_IMPRINT_FORWARD_PARAS 

439 ) 

440 n = len(records) 

441 regions: list[ImprintRegion] = [] 

442 for i, rec in enumerate(records): 

443 if rec.kind != "para" or i in skip_indices: 

444 continue 

445 if imprint_marker(rec.text) is None: 

446 continue 

447 

448 preceding: set[int] = set() 

449 remaining = 2 

450 k = i - 1 

451 while k >= 0 and remaining: 

452 r = records[k] 

453 if _blank_or_skipped(r, k, skip_indices): 

454 k -= 1 

455 continue 

456 if r.kind != "para": 

457 break 

458 preceding.add(k) 

459 remaining -= 1 

460 k -= 1 

461 

462 members: set[int] = {i} 

463 closer: int | None = None 

464 seen = 0 

465 j = i + 1 

466 while j < n and seen < forward_paras: 

467 r = records[j] 

468 if _blank_or_skipped(r, j, skip_indices): 

469 j += 1 

470 continue 

471 if r.kind != "para": 

472 break 

473 seen += 1 

474 members.add(j) 

475 # Only a 印发-family CLOSER ends the region; a start marker (抄送 / 

476 # 主题词) here is middle content, so the walk runs through it. 

477 if imprint_closer(r.text) is not None: 

478 closer = j 

479 break 

480 j += 1 

481 if closer is None: 

482 members = {i} # no closer within the window → anchor only 

483 else: 

484 # Absorb a mis-ordered 成文日期 trailing the closer (see docstring). 

485 date_budget = 2 

486 m = closer + 1 

487 while m < n and date_budget: 

488 r = records[m] 

489 if _blank_or_skipped(r, m, skip_indices): 

490 m += 1 

491 continue 

492 if r.kind != "para" or not document_date(r.text): 

493 break 

494 members.add(m) 

495 date_budget -= 1 

496 m += 1 

497 

498 regions.append( 

499 ImprintRegion(anchor=i, closer=closer, members=members, preceding=preceding) 

500 ) 

501 return regions

506 *,

516 suppressed_events: list[dict] | None = None,

518) -> list[TitleBlockCandidate]: 

519 """Find multi-paragraph windows and single-paragraph title candidates. 

520 

521 ``strong_body`` / ``numbering_veto`` / ``imprint_marker`` / ``imprint_closer`` 

522 / ``document_date`` default to the guardrails implementations and are 

523 injectable for NLP-free tests. ``imprint_excluded`` (the 版记 veto set) 

524 and ``imprint_boundary_indices`` (each region's LAST member — closer or 

525 its absorbed 成文日期 line, the 公文汇编 document-boundary evidence) are 

526 both derived from ONE internal :func:`detect_imprint_regions` call when 

527 either is None, so test and production callers share the same semantics; 

528 ``run_smart_heading`` passes precomputed sets to reuse its regions. 

529 

530 Mid-document gate: a multi/table window whose start lies past 

531 the document HEAD ZONE — more than ``head_zone_records`` CONTENT records 

532 (semantic-text paragraphs / tables; blanks, TOC lines, section breaks 

533 and pure-drawing logo paragraphs don't count) precede it — is only 

534 admitted when it carries document-boundary evidence: its preceding 

535 non-blank record is an imprint-region tail or a bare 附件 marker line,
