1#!/usr/bin/env python3 

2"""Backend-to-backend graph storage migration (Phase 1). 

3 

4Phase 1 migrates a graph from ``PGGraphStorage`` (Apache AGE) to 

5``PGTableGraphStorage`` through the public storage API only: 

6``get_all_nodes`` / ``get_all_edges`` to read, ``upsert_nodes_batch`` / 

7``upsert_edges_batch`` to write, ``remove_nodes`` / ``remove_edges`` to 

8compensate on failure. The first half of this module is pure, 

9side-effect-free building blocks — no I/O, no async, no database imports — 

10so every migration invariant can be unit-tested offline; the second half is 

11the async orchestration (preconditions, write pipeline, verification, 

12id-scoped compensation) driven entirely through the storage objects it is 

13handed, so it stays testable with fakes and importable without asyncpg. 

14 

15Shared edge-dict shape (the "enumerated edge" currency of this module): 

16both backends' ``get_all_edges`` return each edge as a single flat dict of 

17its properties with the endpoint ids merged in under the ``source`` and 

18``target`` keys. The AGE backend enumerates the *directed* edge table and 

19documents that "if 2 directional edges exist between the same pair of nodes, 

20deduplication must be handled by the caller" — this module is that caller. 

21 

22Ordering invariant carried over from ``pgtable_impl``: the target stores 

23each undirected edge once in canonical order ``src_id = min(a, b)``, 

24``tgt_id = max(a, b)`` computed with **Python** ``min``/``max`` (never SQL 

25``LEAST``/``GREATEST``: under non-C collations the two orderings diverge on 

26non-ASCII ids and would produce duplicate edges). ``canonicalize_edge`` is 

27the single place this tool decides canonical order. 

28 

29This tool copies the graph verbatim or refuses; it never silently changes 

30what it migrates. Alongside reciprocal pairs and duplicate node ids it also 

31refuses **self-loop edges**: a graph must not contain one 

32(``BaseGraphStorage.node_degree``), and neither available alternative is safe 

33— carrying it across propagates the violation, while dropping it orphans the 

34matching ``relationships_vdb`` row past the reach of ``adelete_by_relation``. 

35The operator deletes those relations at the source first, where the graph edge 

36still exists so the vector row goes with them. 

37 

38Usage (dry run is the default; no graph data is written without 

39--apply, though initializing the backends still creates their schema): 

40 

41 python -m lightrag.tools.migrate_graph_storage 

42 python -m lightrag.tools.migrate_graph_storage --apply [--workspace WS] 

43 

44Payload comparison is deliberately TYPE-STRICT (see ``payloads_equal``). 

45Residual risk accepted with that choice: if the agtype-to-JSONB round trip 

46ever drifts a numeric representation (e.g. ``1`` read back as ``1.0``), 

47type-strict verification will abort a migration that lost nothing. That 

48failure is visible, diagnosable at the source, and recoverable; the 

49opposite error — blessing a payload that silently changed type — is not. 

50 

51Manual e2e checklist (not testable with fakes): real AGE builds the edge 

52``source``/``target`` ids via ``(agtype_access_operator(...))::text``; 

53whether that cast can carry agtype quoting into the id strings can only be 

54answered against a live AGE instance (cf. the lossy agtype-quote-strip 

55work, same territory). If it ever does, the failure mode here is 

56fail-closed — endpoint ids diverge, verification aborts and compensates — 

57not silent loss. 

58 

59Known caveat: a payload key literally named ``source`` or ``target`` is 

60already destroyed upstream — ``PGGraphStorage.get_all_edges`` overwrites 

61those keys with the endpoint ids when flattening each row — so by the time 

62these helpers run, the original value is unrecoverable and ``edge_properties`` 

63strips the keys as endpoint identity. LightRAG's own edge schema uses 

64``source_id`` for chunk attribution, so this does not arise in practice. 

65"""

83def canonicalize_edge(source: str, target: str) -> tuple[str, str]: 

84 """Return the canonical (src, tgt) order for an undirected edge. 

85

104def payloads_equal(a: Any, b: Any) -> bool: 

105 """Type-strict structural equality for payload values. 

106 

107 The single comparison rule used everywhere this module compares payloads 

108 — ``verify_source_side``, node and edge payloads alike. Recurses into 

109 dicts and lists; every leaf must match in TYPE and value. Bare ``==`` 

110 would coerce across ``bool``/``int``/``float`` (``True == 1``, 

111 ``1 == 1.0``), silently passing a payload whose type changed — and both 

112 agtype and JSON distinguish these types, so the change is observable 

113 downstream. ``type(a) is type(b)`` is required rather than isinstance 

114 checks precisely because ``bool`` subclasses ``int``. 

115 

116 Under the fail-closed policy any observable difference must count as 

117 divergence: a false abort is resolvable by an operator at the source, 

118 a false pass is silent data loss. Consequences, all deliberate: 

119 

120 - key order never matters (dict comparison is key-set + per-key); 

121 - a key present with ``None`` differs from an absent key; 

122 - ``"1"`` vs ``1``, ``True`` vs ``1``, ``1`` vs ``1.0`` all diverge; 

123 - ``NaN`` never equals anything, itself included, so byte-identical 

124 payloads containing ``NaN`` compare unequal — a false positive in the 

125 safe (abort) direction. 

126 

127 Leaf ``==`` is reached ONLY for the exact JSON scalar types (str, int, 

128 float, bool, None). Tuples recurse like lists (positional containers, 

129 so hand-built input cannot smuggle a coercing leaf comparison past the 

130 type gate inside a tuple wrapper); any other type — sets, custom 

131 classes — compares unequal unconditionally, because ``==`` on an 

132 unknown type may coerce and this function must be unfoolable. 

133 JSON/agtype-parsed payloads only ever contain the safe set, so neither 

134 branch fires on real migration data. 

135 """ 

136 if type(a) is not type(b): 

137 return False 

138 if isinstance(a, dict): 

139 return a.keys() == b.keys() and all( 

140 payloads_equal(value, b[key]) for key, value in a.items() 

141 ) 

142 if isinstance(a, (list, tuple)): 

143 return len(a) == len(b) and all(payloads_equal(x, y) for x, y in zip(a, b)) 

144 if isinstance(a, float) and not isinstance(a, bool): 

145 # -0.0 == +0.0 is True, but they are observably different: json.dumps 

146 # renders them differently and math.copysign distinguishes them. Under 

147 # the fail-closed rule an observable difference must count. 

148 import math 

149 

150 return a == b and math.copysign(1.0, a) == math.copysign(1.0, b) 

151 if type(a) in (str, int, float, bool, type(None)): 

152 return a == b 

153 # Unknown type: fail closed rather than trust its __eq__. 

154 return False

157def validate_source_nodes(nodes: Iterable[dict[str, Any]]) -> None: 

158 """Reject malformed enumerated nodes BEFORE anything sorts or groups them. 

159 

160 A real AGE graph can contain a vertex with no properties at all (``CREATE 

161 (n:base)``), which enumerates as ``{"id": None}``. Sorting or grouping that 

162 alongside real ids raises ``TypeError: '<' not supported between instances 

163 of 'NoneType' and 'str'`` — a crash instead of the refusal the operator 

164 needs, and it happens before the entity_id check can speak. So identity is 

165 validated first, and the error names the offending node and the type it 

166 actually had. 

167 

168 ``id`` and ``entity_id`` must both be present, both be ``str``, and agree: 

169 the enumerator derives ``id`` from ``entity_id``, so a disagreement means 

170 the row is not what this tool assumes and is refused defensively. 

171 """ 

172 for position, node in enumerate(nodes): 

173 node_id = node.get("id") 

174 entity_id = node.get("entity_id") 

175 if not isinstance(node_id, str): 

176 raise MigrationDataError( 

177 f"source node at position {position} has a non-string id " 

178 f"({type(node_id).__name__}: {node_id!r}); a vertex with no " 

179 "properties enumerates this way. Repair or remove it before " 

180 "migrating." 

181 ) 

182 if not isinstance(entity_id, str): 

183 raise MigrationDataError( 

184 f"source node {node_id!r} has no usable entity_id " 

185 f"({type(entity_id).__name__}: {entity_id!r}); the target " 

186 "requires one on every node." 

187 ) 

188 if node_id != entity_id: 

189 raise MigrationDataError( 

190 f"source node id {node_id!r} disagrees with its entity_id " 

191 f"{entity_id!r}; refusing rather than guessing which is the " 

192 "identity." 

193 )

196def _is_non_text_safe(text: str) -> bool: 

197 """True when PostgreSQL ``text`` (and therefore jsonb) cannot hold ``text``. 

198 

199 Exactly two cases, both of which a Python ``str`` and AGE's agtype accept 

200 but ``text`` does not: an embedded NUL, and an unpaired surrogate. The 

201 surrogate test is an explicit code-point range rather than a 

202 ``value.encode("utf-8")`` try/except because the range says what is 

203 rejected, while the encode also depends on which surrogate handler the 

204 driver happens to use. 

205 """ 

206 return "\x00" in text or any(0xD800 <= ord(char) <= 0xDFFF for char in text)

209def _unrepresentable_paths(value: Any, path: str = "") -> list[str]: 

210 """Paths of every value PostgreSQL jsonb cannot store faithfully. 

211 

212 Two float families qualify, for the same underlying reason — the target 

213 cannot round-trip them, so the migration would fail or silently change a 

214 value AFTER the point of no return: 

215 

216 - ``NaN`` / ``+-Infinity``: agtype represents them, jsonb rejects them 

217 outright ("invalid input syntax for type json"); 

218 - ``-0.0``: jsonb accepts it but normalises the sign away 

219 (``'{"w": -0.0}'::jsonb`` stores ``{"w": 0.0}``). This tool's comparison 

220 rule deliberately treats signed zeros as different, so migrating one 

221 would write successfully and then fail its own verification. 

222 

223 Two string families qualify for the same reason — jsonb is stored as 

224 PostgreSQL ``text``, which is strictly valid UTF-8 with no embedded NUL: 

225 

226 - a ``\\x00`` anywhere in a string ("unsupported Unicode escape sequence: 

227 \\u0000 cannot be converted to text"); 

228 - a lone surrogate (U+D800..U+DFFF), which Python tolerates in a ``str`` 

229 but which has no UTF-8 encoding at all, so the driver cannot even send 

230 it. AGE stores both happily, so a source graph can carry them. 

231 

232 Dict KEYS are checked with the same string rule: a key is serialized into 

233 the same jsonb document, so a NUL or lone surrogate in a key aborts the 

234 write exactly as one in a value does. 

235 """ 

236 found: list[str] = [] 

237 if isinstance(value, float) and not isinstance(value, bool): 

238 import math 

239 

240 if ( 

241 value != value 

242 or value in (float("inf"), float("-inf")) 

243 or (value == 0.0 and math.copysign(1.0, value) < 0) 

244 ): 

245 found.append(path or "<root>") 

246 elif isinstance(value, str): 

247 if _is_non_text_safe(value): 

248 found.append(path or "<root>") 

249 elif isinstance(value, dict): 

250 for key, item in value.items(): 

251 child = f"{path}.{key}" if path else str(key) 

252 if isinstance(key, str) and _is_non_text_safe(key): 

253 found.append(child) 

254 found.extend(_unrepresentable_paths(item, child)) 

255 elif isinstance(value, (list, tuple)): 

256 for index, item in enumerate(value): 

257 found.extend(_unrepresentable_paths(item, f"{path}[{index}]")) 

258 return found

261def find_non_jsonb_representable( 

262 labelled_payloads: Iterable[tuple[str, dict[str, Any]]], 

263) -> list[tuple[str, list[str]]]: 

264 """Find payloads PostgreSQL ``jsonb`` cannot store faithfully. 

265 

266 The general rule this enforces: if the target cannot round-trip a value, 

267 refuse BEFORE the destructive step. Nothing in the plan phase serializes to 

268 JSONB, so without this check such a payload passes the dry run and only 

269 fails at the write — which under ``--force-empty-target`` is after the 

270 pre-existing slice was dropped, leaving neither the old data nor the new. 

271 The tool already knows it is migrating agtype into jsonb, so a deterministic 

272 failure can be predicted instead of discovered. 

273 

274 See ``_unrepresentable_paths`` for what qualifies and why. 

275 """ 

276 offenders = [] 

277 for label, payload in labelled_payloads: 

278 paths = _unrepresentable_paths(payload) 

279 if paths: 

280 offenders.append((label, sorted(paths))) 

281 return sorted(offenders)

284def detect_reciprocal_pairs( 

285 directed_edges: Iterable[dict[str, Any]], 

286) -> list[tuple[str, str]]: 

287 """Canonical keys that have BOTH directions present, payloads irrelevant. 

288 

289 Phase 1 refuses every reciprocal pair, not only the ones whose payloads 

290 disagree. Even an identical ``a->b`` / ``b->a`` pair is two relationship 

291 rows in AGE and one row in the target, and AGE computes degree by counting 

292 relationship rows — so collapsing them changes degree and traversal even 

293 though no property is lost. LightRAG's own write path matches edges 

294 undirected (``OPTIONAL MATCH (source)-[old:DIRECTED]-(target)``), so it does 

295 not produce reciprocals: finding one means the source violates the 

296 invariant this tool assumes, and refusing keeps ``verified`` meaning that 

297 node/edge cardinality and degree were preserved too. 

298 

299 A payload-aware relaxation — rejecting only the reciprocal pairs whose two 

300 payloads disagree — is deliberately NOT provided. It would be wrong, not 

301 merely conservative: AGE computes degree by counting relationship rows, so 

302 collapsing even a byte-identical ``a->b`` / ``b->a`` pair drops the degree 

303 of both endpoints by one and changes what traversal returns, while losing 

304 no property. Phase 1 therefore fails closed on all of them, and an operator 

305 who knows the reciprocal is redundant deletes it at the source instead. 

306 """ 

307 seen: dict[tuple[str, str], set[tuple[str, str]]] = {} 

308 for edge in directed_edges: 

309 src, tgt = edge["source"], edge["target"] 

310 if src == tgt: 

311 continue # a self-loop has only one orientation 

312 seen.setdefault(canonicalize_edge(src, tgt), set()).add((src, tgt)) 

313 return sorted(key for key, directions in seen.items() if len(directions) > 1)

316def detect_self_loop_edges( 

317 directed_edges: Iterable[dict[str, Any]], 

318) -> list[str]: 

319 """Node ids carrying a ``src == tgt`` edge, sorted. 

320 

321 ``BaseGraphStorage.node_degree`` states that a graph must not contain a 

322 self-loop: every LightRAG ingress refuses one, and degree is not contracted 

323 on it precisely because no admissible graph can hold one. So the source is 

324 invariant-violating and this tool refuses it, exactly as it refuses 

325 reciprocal pairs and duplicate node ids. 

326 

327 **Refused rather than dropped, and the reason is cross-store.** A self-loop 

328 old enough to predate the ``ainsert_custom_kg`` guard also has a 

329 ``relationships_vdb`` row. This tool migrates the GRAPH only, so silently 

330 omitting the edge would orphan that row: ``_get_edge_data`` would keep 

331 retrieving the stale vector hit and spending a ``top_k`` slot before 

332 discarding it for having no graph edge, and -- worse -- 

333 ``adelete_by_relation`` 404s when ``has_edge`` is false, so the row would 

334 become unreachable through every public API. Refusing keeps the cleanup 

335 possible: run it against the SOURCE, where the graph edge still exists, and 

336 the relation and its vector row go together. 

337 """ 

338 return sorted( 

339 {edge["source"] for edge in directed_edges if edge["source"] == edge["target"]} 

340 )

343def detect_duplicate_node_ids(nodes: Iterable[dict[str, Any]]) -> list[str]: 

344 """Node ids enumerated more than once, payloads irrelevant. 

345 

346 Same reasoning as reciprocals: the target's primary key keeps one row per 

347 id, so two physical source vertices sharing an entity_id become one node 

348 and the graph's node cardinality (and any degree derived from the edges 

349 that pointed at each of them) changes. Refused regardless of whether the 

350 payloads match. 

351 

352 As with ``detect_reciprocal_pairs``, a payload-aware relaxation that passed 

353 duplicates with identical payloads is deliberately NOT provided: the 

354 cardinality and degree change happens whether or not a property differs, so 

355 equal payloads make the collapse invisible rather than harmless. 

356 """ 

357 counts: dict[str, int] = {} 

358 for node in nodes: 

359 node_id = node.get("id") 

360 if isinstance(node_id, str): 360 ↛ 358line 360 didn't jump to line 358 because the condition on line 360 was always true

361 counts[node_id] = counts.get(node_id, 0) + 1 

362 return sorted(node_id for node_id, count in counts.items() if count > 1)

365def count_parallel_edges( 

366 directed_edges: Iterable[dict[str, Any]], 

367) -> dict[tuple[str, str], int]: 

368 """Count enumerated rows per ORDERED (source, target) pair. 

369 

370 Backstop for the AGE invariant "at most one directed edge per ordered 

371 pair": a count above 1 means parallel edges survived the AGE enumeration 

372 query's SELECT DISTINCT (over source, target and properties), so their 

373 agtype representations differ — though the parsed Python payloads may 

374 still compare equal (e.g. a ``1`` vs ``1.0`` representation drift), so a 

375 violation is not proof of semantic divergence. Canonicalization would 

376 keep only one row either way; the caller fails closed on any count > 1. 

377 

378 Keyed by the ordered pair, not the canonical key, because a legitimate 

379 reciprocal pair (a->b plus b->a) also maps two rows onto one canonical 

380 key — that case belongs to ``detect_reciprocal_pairs``, not here. 

381 """ 

382 counts: dict[tuple[str, str], int] = {} 

383 for edge in directed_edges: 

384 key = (edge["source"], edge["target"]) 

385 counts[key] = counts.get(key, 0) + 1 

386 return counts

389def derive_written_node_ids( 

390 migrated_node_ids: Iterable[str], 

391 edges: Iterable[dict[str, Any]], 

392) -> set[str]: 

393 """The exact node-id set a target-side compensation must remove. 

394 

395 ``written = migrated_node_ids UNION {every edge endpoint}``. 

396 

397 ``PGTableGraphStorage.upsert_edges_batch`` returns ``None`` and silently 

398 auto-creates missing endpoints (a CTE inserting stub nodes with 

399 ``ON CONFLICT DO NOTHING``), so endpoints that were never in the node 

400 batch are written without being observable in any return value — they 

401 must be DERIVED here from the edge list itself. 

402 

403 Precondition (empty-target invariant): this set equals "everything the 

404 migration wrote" ONLY if the target workspace/namespace slice was empty 

405 when the migration started. On a non-empty target, ``ON CONFLICT`` paths 

406 touch pre-existing rows this set cannot distinguish from migration 

407 writes, so an id-scoped compensation keyed on it would delete data the 

408 migration never created. 

409 """ 

410 written = set(migrated_node_ids) 

411 for edge in edges: 

412 written.add(edge["source"]) 

413 written.add(edge["target"]) 

414 return written

417def sort_directed_edges( 

418 directed_edges: Iterable[dict[str, Any]], 

419) -> list[dict[str, Any]]: 

420 """Deterministically order an enumerated directed edge list. 

421 

422 The AGE enumeration query has no ORDER BY, so two reads of the same 

423 graph may enumerate in different orders; every downstream step must see 

424 a stable order for two runs to write and report identically. Sorted by 

425 (source, target) with a canonical-JSON rendering of the payload as 

426 tiebreaker, so the order is total even on invariant-violating input 

427 (parallel rows), which fails closed later with a deterministic message. 

428 

429 The tiebreaker serialisation can raise ``TypeError`` on a nested payload 

430 dict with mixed-type keys and ``ValueError`` on a circular payload. 

431 Neither can arise from JSON-parsed AGE payloads (JSON object keys are 

432 always strings, parsed structures are acyclic); on hand-built input the 

433 exception is the correct fail-closed outcome. 

434 """ 

435 return sorted( 

436 directed_edges, 

437 key=lambda edge: ( 

438 edge["source"], 

439 edge["target"], 

440 json.dumps( 

441 edge_properties(edge), sort_keys=True, ensure_ascii=False, default=str 

442 ), 

443 ), 

444 )

447@dataclass 

448class VerificationReport: 

449 """Source-driven post-migration comparison, one field per mismatch class. 

450 

451 Verification is driven from the SOURCE side because the target has 

452 already canonicalized: a payload lost to reciprocal collapse or a 

453 dropped row is invisible when walking the target alone. 

454 """ 

455 

456 # Node-id set differences (source ids vs target ids). 

457 missing_node_ids: list[str] = field(default_factory=list) 

458 unexpected_node_ids: list[str] = field(default_factory=list) 

459 # Node ids present on both sides whose property dicts differ: 

460 # (node_id, source_properties, target_properties). 

461 node_property_mismatches: list[tuple[str, dict[str, Any], dict[str, Any]]] = field( 

462 default_factory=list 

463 ) 

464 # Canonical-edge set differences. 

465 missing_edges: list[tuple[str, str]] = field(default_factory=list) 

466 unexpected_edges: list[tuple[str, str]] = field(default_factory=list) 

467 # Canonical keys present on both sides whose payloads differ: 

468 # (canonical_key, source_properties, target_properties). 

469 edge_property_mismatches: list[ 

470 tuple[tuple[str, str], dict[str, Any], dict[str, Any]] 

471 ] = field(default_factory=list) 

472 # Divergent reciprocals found in the SOURCE enumeration itself. Non-empty 

473 # means the fail-closed precondition was violated upstream; the source 

474 # cannot be compared meaningfully until resolved. 

475 source_reciprocal_pairs: list[tuple[str, str]] = field(default_factory=list) 

476 # Ordered pairs enumerated more than once in the SOURCE (parallel rows), 

477 # with their counts. Same reasoning as reciprocal divergence: a source 

478 # violating the at-most-one-row-per-ordered-pair invariant has no 

479 # well-defined expected target, and canonicalizing its edge map keeps an 

480 # arbitrary survivor — so verification must fail regardless of whether 

481 # the caller already ran the backstop. 

482 source_parallel_violations: dict[tuple[str, str], int] = field(default_factory=dict) 

483 # Node ids enumerated more than once in the SOURCE with differing 

484 # payloads. Same last-line-of-defense reasoning: the per-id maps below 

485 # collapse duplicates last-write-wins, so this is only detectable on the 

486 # raw enumeration — and without it a lost duplicate payload would verify 

487 # clean. 

488 source_duplicate_node_ids: list[str] = field(default_factory=list)

520def verify_source_side( 

521 source_nodes: list[dict[str, Any]], 

522 source_edges: list[dict[str, Any]], 

523 target_nodes: list[dict[str, Any]], 

524 target_edges: list[dict[str, Any]], 

525) -> VerificationReport: 

526 """Compare a migrated target against its source, source-driven. 

527 

528 Inputs are the raw ``get_all_nodes`` / ``get_all_edges`` results of each 

529 backend (nodes carry ``id``; edges carry ``source``/``target`` merged 

530 over the payload; source edges may be AGE-shaped directed rows). 

531 

532 Node ids are compared as sets, then per-id payloads with the shared 

533 type-strict ``payloads_equal`` rule; edges as canonical-key sets, then 

534 per-key payloads under the same rule. All three source-side 

535 preconditions — no divergent reciprocals, no parallel rows per ordered 

536 pair, no divergent duplicate node ids — are 

537 re-checked here and reported rather than assumed, because verification 

538 is the last line of defense: a source that fails its own precondition 

539 has no well-defined expected target, and it must not depend on the 

540 caller having run the backstops. The source enumeration is 

541 deterministically ordered first, so the whole report (not just ``ok``) 

542 is independent of SOURCE enumeration order. (Target order needs no such 

543 pinning: the PGTable primary key (workspace, namespace, src_id, tgt_id) 

544 guarantees at most one row per canonical key, so a real target cannot 

545 present the duplicate rows that would make last-write-wins order- 

546 sensitive.) ``unexpected_*`` entries are 

547 target rows with no source counterpart — under the empty-target 

548 invariant those can only be migration bugs (e.g. stub endpoints for 

549 edges the source never had). 

550 """ 

551 report = VerificationReport() 

552 

553 # AGE enumerates with no ORDER BY; pin the order so map construction 

554 # (last-write-wins on violating input) and every derived field are 

555 # reproducible across runs. Nodes get the same treatment as edges: on 

556 # duplicate-id input the per-id map below keeps the last payload, which 

557 # must not depend on enumeration order. 

558 source_edges = sort_directed_edges(source_edges) 

559 source_nodes = sorted( 

560 source_nodes, 

561 key=lambda node: ( 

562 node["id"], 

563 json.dumps( 

564 _node_payload(node), sort_keys=True, ensure_ascii=False, default=str 

565 ), 

566 ), 

567 ) 

568 

569 # Same detectors the plan phase uses, so verification re-runs the source 

570 # checks rather than an older, weaker version of them: duplicates and 

571 # reciprocals count whether or not their payloads agree, because both 

572 # change cardinality on the way into the target. 

573 report.source_duplicate_node_ids = detect_duplicate_node_ids(source_nodes) 

574 report.source_reciprocal_pairs = detect_reciprocal_pairs(source_edges) 

575 report.source_parallel_violations = { 

576 pair: count 

577 for pair, count in sorted(count_parallel_edges(source_edges).items()) 

578 if count > 1 

579 } 

580 

581 source_node_map = {node["id"]: _node_payload(node) for node in source_nodes} 

582 target_node_map = {node["id"]: _node_payload(node) for node in target_nodes} 

583 report.missing_node_ids = sorted(source_node_map.keys() - target_node_map.keys()) 

584 report.unexpected_node_ids = sorted(target_node_map.keys() - source_node_map.keys()) 

585 for node_id in sorted(source_node_map.keys() & target_node_map.keys()): 

586 if not payloads_equal(source_node_map[node_id], target_node_map[node_id]): 

587 report.node_property_mismatches.append( 

588 (node_id, source_node_map[node_id], target_node_map[node_id]) 

589 ) 

590 

591 source_edge_map = _canonical_edge_map(source_edges) 

592 target_edge_map = _canonical_edge_map(target_edges) 

593 report.missing_edges = sorted(source_edge_map.keys() - target_edge_map.keys()) 

594 report.unexpected_edges = sorted(target_edge_map.keys() - source_edge_map.keys()) 

595 for key in sorted(source_edge_map.keys() & target_edge_map.keys()): 

596 if not payloads_equal(source_edge_map[key], target_edge_map[key]): 

597 report.edge_property_mismatches.append( 

598 (key, source_edge_map[key], target_edge_map[key]) 

599 ) 

600 

601 return report

694def check_migration_pair(source: Any, target: Any) -> None: 

695 """Reject any (source, target) pair outside the Phase 1 allowlist. 

696 

697 Runs before every other precondition and therefore before any read or 

698 write on either storage. 

699 """ 

700 pair = (type(source).__name__, type(target).__name__) 

701 if pair not in ALLOWED_MIGRATION_PAIRS: 

702 allowed = ", ".join(f"{s} -> {t}" for s, t in sorted(ALLOWED_MIGRATION_PAIRS)) 

703 raise MigrationPreconditionError( 

704 f"unsupported migration pair {pair[0]} -> {pair[1]}; " 

705 f"Phase 1 supports exactly: {allowed}" 

706 )

709async def target_slice_is_empty(target: Any) -> bool: 

710 """Empty-target gate, via ``get_all_nodes``. 

711 

712 NEVER ``is_empty()``: graph storages do not have that method at all — 

713 the None-returning stub in ``lightrag/base.py`` belongs to 

714 ``BaseKVStorage``, a different branch of the hierarchy, and 

715 ``BaseGraphStorage`` defines nothing like it. A gate written against it 

716 would crash with AttributeError on a real backend (fail-closed by 

717 accident, and for the wrong reason); this gate asks a question the 

718 graph API actually answers. 

719 """ 

720 return len(await target.get_all_nodes()) == 0

723async def compensate_partial_migration( 

724 target: Any, 

725 written_node_ids: Iterable[str], 

726 written_edge_keys: Iterable[tuple[str, str]], 

727) -> None: 

728 """Remove exactly what this run wrote: edges first, then nodes. 

729 

730 Edge-before-node order follows the FK direction (edges reference their 

731 endpoint rows). Both remove_* contracts are DELETE ... = ANY(ids), so 

732 ids that were never actually written — e.g. when the failure hit before 

733 the edge batch — are silent no-ops, which is why the full derived 

734 superset is always passed. Exact under the empty-target invariant only: 

735 on a target that was not empty at T0, these ids may name pre-existing 

736 rows the migration never created (see derive_written_node_ids). 

737 """ 

738 await target.remove_edges(sorted(written_edge_keys)) 

739 await target.remove_nodes(sorted(written_node_ids))

742async def migrate_graph( 

743 source: Any, target: Any, *, force_empty_target: bool = False 

744) -> MigrationRunReport: 

745 """Migrate the whole graph from source to target via the storage API. 

746 

747 Pipeline: pair allowlist -> empty-target gate -> enumerate + fail-closed 

748 source checks (parallel rows, divergent reciprocals, entity_id present) 

749 -> upsert nodes -> upsert edges -> source-driven verification. Any 

750 failure after writes begin triggers id-scoped compensation and raises 

751 MigrationWriteError carrying the run report. 

752 

753 ``force_empty_target=True`` (CLI: --force-empty-target) is the ONLY path 

754 that touches pre-existing target data: it drops the whole target slice 

755 instead of rejecting a non-empty target. It is never used for 

756 compensation. 

757 """ 

758 check_migration_pair(source, target) 

759 

760 target_non_empty = not await target_slice_is_empty(target) 

761 if target_non_empty and not force_empty_target: 

762 raise MigrationPreconditionError( 

763 "target graph slice is not empty; refusing to write " 

764 "(pass force_empty_target to drop it instead)" 

765 ) 

766 

767 # Plan BEFORE dropping. _plan_migration is read-only and runs every 

768 # fail-closed source check, so ordering it first means a source that fails 

769 # one of them cannot cost the operator their existing target slice: the 

770 # single destructive action happens only once the migration is known to be 

771 # able to proceed. 

772 source_nodes, source_edges, node_items, edge_items, report = await _plan_migration( 

773 source, target 

774 ) 

775 report.resolved_workspace = getattr(target, "workspace", None) 

776 report.target_was_non_empty = target_non_empty 

777 

778 if target_non_empty: 

779 # PGTableGraphStorage.drop() reports failure by RETURNING 

780 # {"status": "error"} instead of raising, so an unchecked call would 

781 # record a slice as dropped while its rows are still there — and the 

782 # migration would then write into, and later compensate against, data 

783 # it does not own. 

784 drop_result = await target.drop() 

785 # Only an explicit success counts. The backend signals failure by 

786 # RETURNING {"status": "error"}, but treating "anything that is not 

787 # error" as success would also bless None, {"status": "failed"} or a 

788 # missing status — after which the run writes into, and compensates 

789 # against, rows it does not own. Phase 1 allow-lists exactly one target, 

790 # so its contract can be required exactly. 

791 if not ( 

792 isinstance(drop_result, dict) and drop_result.get("status") == "success" 

793 ): 

794 raise MigrationPreconditionError( 

795 "failed to drop the target graph slice; expected " 

796 f'{{"status": "success"}} but got {drop_result!r}' 

797 ) 

798 report.dropped_target_slice = True 

799 

800 try: 

801 await target.upsert_nodes_batch(node_items) 

802 await target.upsert_edges_batch(edge_items) 

803 report.verification = verify_source_side( 

804 source_nodes, 

805 source_edges, 

806 await target.get_all_nodes(), 

807 await target.get_all_edges(), 

808 ) 

809 if not report.verification.ok: 

810 raise MigrationError("source-driven verification failed") 

811 except BaseException as exc: 

812 # BaseException, not Exception: asyncio.CancelledError does NOT subclass 

813 # Exception, so Ctrl-C or a cancelled task partway through the writes 

814 # would otherwise leave the target populated with no compensation and no 

815 # report — the one failure mode the id-scoped undo exists to prevent. 

816 try: 

817 await compensate_partial_migration( 

818 target, report.written_node_ids, report.written_edge_keys 

819 ) 

820 report.compensated = True 

821 except Exception as compensation_exc: # noqa: BLE001 — report, then surface the original failure 

822 report.compensation_error = repr(compensation_exc) 

823 if isinstance(exc, asyncio.CancelledError): 

824 # Compensate, then let cancellation keep propagating as itself: 

825 # converting it would break the caller's cancellation semantics. 

826 raise 

827 raise MigrationWriteError( 

828 f"migration failed after writes began ({exc}); " 

829 + ( 

830 "id-scoped compensation completed" 

831 if report.compensated 

832 else "COMPENSATION INCOMPLETE — remove the reported " 

833 "written_edge_keys then written_node_ids by hand" 

834 ), 

835 report, 

836 ) from exc 

837 

838 report.verified = True 

839 return report

842async def _plan_migration( 

843 source: Any, target: Any 

844) -> tuple[ 

845 list[dict[str, Any]], 

846 list[dict[str, Any]], 

847 list[tuple[str, dict[str, Any]]], 

848 list[tuple[str, str, dict[str, Any]]], 

849 MigrationRunReport, 

850]: 

851 """Enumerate the source, run every fail-closed check, build the write 

852 plan. Read-only; shared verbatim by migrate_graph and dry_run_migration 

853 so the dry run exercises exactly the checks the apply run would.""" 

854 source_nodes = await source.get_all_nodes() 

855 # Identity FIRST: a vertex with no properties enumerates as {"id": None}, 

856 # and sorting or grouping that alongside real ids raises TypeError before 

857 # any check can report anything useful. 

858 validate_source_nodes(source_nodes) 

859 

860 source_edges = sort_directed_edges(await source.get_all_edges()) 

861 

862 self_loops = detect_self_loop_edges(source_edges) 

863 if self_loops: 

864 raise MigrationDataError( 

865 "source contains self-loop edges, which a graph must not hold " 

866 "(BaseGraphStorage.node_degree); migrating them forward would " 

867 "carry the violation into the target, and dropping them would " 

868 "orphan the matching relationships_vdb rows beyond the reach of " 

869 "adelete_by_relation. Delete these relations at the SOURCE first, " 

870 "through LightRAG so the vector rows go with them, then re-run: " 

871 f"{self_loops}" 

872 ) 

873 

874 violations = { 

875 pair: count 

876 for pair, count in count_parallel_edges(source_edges).items() 

877 if count > 1 

878 } 

879 if violations: 

880 raise MigrationDataError( 

881 f"source violates the one-directed-edge-per-ordered-pair " 

882 f"invariant; parallel rows for: {sorted(violations)}" 

883 ) 

884 # Reciprocals and duplicate ids are refused whether or not their payloads 

885 # agree. Collapsing them loses no property but does change node/edge 

886 # cardinality and degree, and `verified` is meant to mean the graph came 

887 # across — not merely its payload set. 

888 reciprocals = detect_reciprocal_pairs(source_edges) 

889 if reciprocals: 

890 raise MigrationDataError( 

891 "source has reciprocal directed edges (a->b and b->a); the target " 

892 "stores one row per canonical pair, so degree and traversal would " 

893 f"change even where payloads match: {reciprocals}" 

894 ) 

895 duplicate_nodes = detect_duplicate_node_ids(source_nodes) 

896 if duplicate_nodes: 

897 raise MigrationDataError( 

898 "source enumerates the same node id more than once; the target " 

899 "keeps one row per id, so node cardinality would change even " 

900 f"where payloads match: {duplicate_nodes}" 

901 ) 

902 non_jsonb = find_non_jsonb_representable( 

903 [(f"node {n['id']!r}", _node_payload(n)) for n in source_nodes] 

904 + [ 

905 ( 

906 f"edge {canonicalize_edge(e['source'], e['target'])!r}", 

907 edge_properties(e), 

908 ) 

909 for e in source_edges 

910 ] 

911 ) 

912 if non_jsonb: 

913 raise MigrationDataError( 

914 "source payloads contain values PostgreSQL jsonb cannot store " 

915 "faithfully (NaN, +-Infinity, -0.0 whose sign jsonb normalises " 

916 "away, or a string carrying a NUL or an unpaired surrogate, " 

917 "neither of which PostgreSQL text can hold); migrating them would " 

918 "fail after the point of no return: " 

919 f"{[(label, paths) for label, paths in non_jsonb]}" 

920 ) 

921 

922 node_items: list[tuple[str, dict[str, Any]]] = [] 

923 for node in source_nodes: 

924 payload = _node_payload(node) 

925 if "entity_id" not in payload: 925 ↛ 926line 925 didn't jump to line 926 because the condition on line 925 was never true

926 raise MigrationDataError( 

927 f"source node {node.get('id')!r} has no entity_id property; " 

928 "the target backend requires it" 

929 ) 

930 node_items.append((node["id"], payload)) 

931 

932 canonical_edges = _canonical_edge_map(source_edges) 

933 edge_items = [ 

934 (src, tgt, props) for (src, tgt), props in sorted(canonical_edges.items()) 

935 ] 

936 

937 report = MigrationRunReport( 

938 source_backend=type(source).__name__, 

939 target_backend=type(target).__name__, 

940 node_count=len(node_items), 

941 edge_count=len(edge_items), 

942 # Derived before the first write so a failure at ANY later point 

943 # already has the complete compensation set, auto-created edge 

944 # endpoints included. 

945 written_node_ids=sorted( 

946 derive_written_node_ids( 

947 (node_id for node_id, _ in node_items), source_edges 

948 ) 

949 ), 

950 written_edge_keys=sorted(canonical_edges), 

951 ) 

952 return source_nodes, source_edges, node_items, edge_items, report

955async def dry_run_migration( 

956 source: Any, target: Any, *, force_empty_target: bool = False 

957) -> MigrationRunReport: 

958 """Read-only preflight: everything migrate_graph checks, zero writes. 

959 

960 Runs the pair allowlist, the empty-target gate and every fail-closed 

961 source check, and returns the report the apply run would start from 

962 (counts and the would-be written id sets, ``verified`` False, no 

963 verification). Raises exactly the errors the apply run would raise at 

964 the same points. With ``force_empty_target`` a non-empty target is not 

965 rejected — but it is NOT dropped either; the drop happens only in the 

966 apply run. 

967 """ 

968 check_migration_pair(source, target) 

969 target_non_empty = not await target_slice_is_empty(target) 

970 if target_non_empty and not force_empty_target: 

971 # Identical wording to migrate_graph: same code path, same claim. 

972 raise MigrationPreconditionError( 

973 "target graph slice is not empty; refusing to write " 

974 "(pass force_empty_target to drop it instead)" 

975 ) 

976 _, _, _, _, report = await _plan_migration(source, target) 

977 # Reaching here with a non-empty target means force mode: the apply run 

978 # WILL drop the pre-existing slice. The report must say so — a 

979 # clean-looking force-mode preview of a destructive apply is exactly the 

980 # operator trap this field exists to prevent. 

981 report.resolved_workspace = getattr(target, "workspace", None) 

982 report.target_was_non_empty = target_non_empty 

983 return report

991def _precheck_requested_workspace(requested: str) -> None: 

992 """Refuse an env-overridden workspace BEFORE anything is constructed. 

993 

994 ``initialize()`` is not read-only: ``PGTableGraphStorage`` runs its DDL and 

995 then ``_normalize_legacy_edges``, which DELETEs and re-INSERTs edge rows. 

996 So a check that can only read the *resolved* workspace — which the backend 

997 settles during ``initialize()`` — necessarily runs after the wrong slice has 

998 already been touched. This mirrors the backend's documented priority 

999 (``POSTGRES_WORKSPACE`` env > ``config.ini`` ``[postgres] workspace`` > 

1000 constructor argument > ``"default"``, pgtable_impl.py:531 resolving what 

1001 ``PGPostgresDB.get_config`` settled) to catch the dangerous case before any 

1002 connection is opened. ``_check_resolved_workspace`` stays as the backstop 

1003 for whatever this cannot predict. 

1004 

1005 ``config.ini`` matters as much as the env var and is easier to forget: it 

1006 is not visible in the operator's shell, so a checked-in 

1007 ``[postgres] workspace`` silently outranks ``--workspace`` with nothing on 

1008 screen to suggest it. The error names which of the two sources spoke. 

1009 """ 

1010 env_workspace = os.getenv("POSTGRES_WORKSPACE") 

1011 if env_workspace is not None: 

1012 # An env var set to "" is what get_config passes on, and an empty 

1013 # workspace is falsy to the backend, so the constructor argument still 

1014 # wins — and config.ini is never consulted. Mirror that exactly. 

1015 effective, origin = env_workspace, "POSTGRES_WORKSPACE" 

1016 else: 

1017 # Same read as PGPostgresDB.get_config: cwd-relative, and a missing 

1018 # file makes ConfigParser.read a silent no-op. 

1019 config = configparser.ConfigParser() 

1020 config.read("config.ini", "utf-8") 

1021 effective = config.get("postgres", "workspace", fallback=None) or "" 

1022 origin = "config.ini [postgres] workspace" 

1023 if requested and effective and effective != requested: 

1024 raise MigrationPreconditionError( 

1025 f"requested workspace {requested!r} but {origin} is set to " 

1026 f"{effective!r} and outranks it; refusing before opening any " 

1027 "connection, because initializing a graph storage mutates the " 

1028 "workspace it lands in" 

1029 )

1032def _check_resolved_workspace(requested: str, source: Any, target: Any) -> None: 

1033 """Fail closed when the backends did not land on the workspace we asked for. 

1034 

1035 The graph backends resolve their workspace as ``POSTGRES_WORKSPACE`` env > 

1036 ``config.ini`` ``[postgres] workspace`` > the value passed to the 

1037 constructor > ``"default"`` (pgtable_impl.py:531 on top of 

1038 ``PGPostgresDB.get_config``), so the environment or a checked-in config 

1039 file silently outranks ``--workspace``. That matters 

1040 because this tool has a destructive mode: without this check, 

1041 ``--workspace staging --force-empty-target`` could drop a slice the 

1042 operator never named. An empty request means the operator named nothing, so 

1043 any resolution is legitimate. 

1044 """ 

1045 resolved_source = getattr(source, "workspace", None) 

1046 resolved_target = getattr(target, "workspace", None) 

1047 if resolved_source != resolved_target: 

1048 raise MigrationPreconditionError( 

1049 "source and target resolved to different workspaces " 

1050 f"({resolved_source!r} vs {resolved_target!r}); refusing to migrate" 

1051 ) 

1052 if requested and resolved_target != requested: 

1053 raise MigrationPreconditionError( 

1054 f"requested workspace {requested!r} but the backends resolved to " 

1055 f"{resolved_target!r} (POSTGRES_WORKSPACE overrides --workspace); " 

1056 "refusing to operate on a workspace that was not named" 

1057 )

1060async def _build_graph_storage(backend_name: str, workspace: str) -> Any: 

1061 """Instantiate and initialize one graph storage via the factory. 

1062 

1063 Imports are deliberately INSIDE the function (the rebuild_vdb pattern): 

1064 the offline CI does not install asyncpg, and importing this module or 

1065 its tests must never pull in a DB backend — only actually running the 

1066 CLI may. 

1067 """ 

1068 from lightrag.kg.factory import get_storage_class 

1069 from lightrag.namespace import NameSpace 

1070 from lightrag.utils import EmbeddingFunc 

1071 

1072 async def _never_embed(*_args: Any, **_kwargs: Any) -> Any: 

1073 raise RuntimeError("graph migration never embeds") 

1074 

1075 cls = get_storage_class(backend_name) 

1076 storage = cls( 

1077 namespace=NameSpace.GRAPH_STORE_CHUNK_ENTITY_RELATION, 

1078 workspace=workspace, 

1079 # Graph storages read their configuration via global_config.get(...) 

1080 # (connection settings come from POSTGRES_* env vars); nothing is 

1081 # required here for the graph-only paths this tool uses. 

1082 global_config={},
