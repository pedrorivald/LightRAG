1"""Admission ticket handed from the ASGI middleware to the route (LR2 §9.3). 

2 

3The capacity decision has to happen before the first ``receive()`` — a request 

4that will be refused must never have its body read — but the reservation it 

5takes has to stay held until the documents are visible in ``doc_status``, which 

6happens in a background task long after the response is sent. The ticket is how 

7that one reservation travels across the three owners: 

8 

9 ASGI middleware → endpoint adopt → managed background task → release 

10 

11Exactly one owner is responsible at any moment, and each hands over explicitly: 

12 

13* not adopted (rejected before the route, or the route never ran) — the 

14 middleware releases it in its ``finally``; 

15* adopted, not handed off (validation rejection, streaming failure) — the 

16 endpoint releases it in its ``finally``; 

17* handed off — the background task's ``finally`` / backstop releases it. 

18 

19Returning the HTTP response must NOT release: the client has been told the 

20document was accepted, so the reservation stays charged until it lands. 

21 

22This module is intentionally dependency-free so both the middleware and the 

23document routes can import it without a cycle. 

24""" 

25 

26from __future__ import annotations 

27 

28from dataclasses import dataclass 

29from typing import Any, Optional 

30 

31# Key under which the ticket is published in the ASGI ``scope["state"]`` dict. 

32# Starlette exposes that dict as ``request.state``, which is how the route reads 

33# it without the middleware having to know anything about FastAPI. 

34ADMISSION_STATE_KEY = "lightrag_admission_ticket" 

35 

36 

37@dataclass 

38class AdmissionTicket: 

39 """One pending-enqueue reservation in flight across layers.""" 

40 

41 token: str 

42 weight: int = 1 

43 adopted: bool = False 

44 

45 

46def publish_admission_ticket(scope: dict[str, Any], ticket: AdmissionTicket) -> None: 

47 """Put ``ticket`` where the route can find it.""" 

48 scope.setdefault("state", {})[ADMISSION_STATE_KEY] = ticket 

49 

50 

51def adopt_admission_ticket(request: Any) -> Optional[AdmissionTicket]: 

52 """Take ownership of the middleware's reservation, if there is one. 

53 

54 Returns the ticket (now marked adopted, so the middleware's ``finally`` 

55 leaves it alone) or ``None`` when no middleware ran for this request — the 

56 route then takes its own reservation exactly as before, which is what keeps 

57 admission correct when the middleware is not installed (capacity disabled) 

58 or when the route is exercised directly by a test rig. 

59 """ 

60 state = getattr(request, "state", None) 

61 ticket = getattr(state, ADMISSION_STATE_KEY, None) if state is not None else None 

62 if not isinstance(ticket, AdmissionTicket): 

63 return None 

64 ticket.adopted = True 

65 return ticket
