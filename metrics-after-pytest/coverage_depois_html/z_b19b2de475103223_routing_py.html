290def slim_chunk_options( 

291 snapshot: Mapping[str, Any] | None, 

292 process_options: Any = "", 

293) -> dict[str, Any]: 

294 """Project a (possibly full) chunker snapshot down to the active strategy. 

295 

296 Keeps the top-level ``chunk_token_size`` and the one strategy 

297 sub-dict picked by :func:`chunk_strategy_key`; everything else is 

298 discarded. Idempotent: a slim snapshot whose key already matches 

299 ``process_options`` passes through unchanged (deep-copied for 

300 isolation). When the matching strategy block is absent from the 

301 input, an empty dict is used so downstream consumers always see a 

302 dict-shaped slot. 

303 

304 Strategy-specific default backfill: for ``paragraph_semantic`` we 

305 guarantee a populated ``chunk_token_size`` slot before returning 

306 (caller-supplied value > ``CHUNK_P_SIZE`` env > 

307 ``DEFAULT_CHUNK_P_SIZE``). This is the single chokepoint that 

308 every enqueue path runs through — both the 

309 ``resolve_chunk_options`` path (built from addon_params) AND the 

310 direct ``chunk_options=`` kwarg path (caller supplies the dict) 

311 flow through here, so the backfill cannot be bypassed by runtime 

312 addon_params mutation or by passing an explicit ``chunk_options`` 

313 that omits the P slot. P must NOT inherit the top-level 

314 ``chunk_token_size`` (global ``CHUNK_SIZE`` / legacy ctor) — 

315 paragraph-semantic merging needs more headroom than the global 

316 default. 

317 """ 

318 key = chunk_strategy_key(process_options) 

319 src: Mapping[str, Any] = snapshot or {} 

320 result: dict[str, Any] = {} 

321 if "chunk_token_size" in src: 

322 result["chunk_token_size"] = deepcopy(src["chunk_token_size"]) 

323 result[key] = deepcopy(dict(src.get(key) or {})) 

324 if key == "paragraph_semantic": 

325 if "chunk_token_size" not in result[key]: 

326 result[key]["chunk_token_size"] = _chunk_env_int( 

327 "CHUNK_P_SIZE", DEFAULT_CHUNK_P_SIZE 

328 ) 

329 # Mirror the CHUNK_P_DROP_REFERENCES env default for the drop-references 

330 # switch here — this is the single chokepoint every enqueue path runs 

331 # through, so a runtime ``addon_params`` mutation or an explicit 

332 # ``chunk_options=`` that omits the slot still picks up the env default. 

333 # ``setdefault`` keeps any caller-supplied value (hint/addon/kwarg). The 

334 # detection-tuning knobs (tail window / heading prefixes) are NOT 

335 # snapshotted — the chunker reads them live from env at run time. 

336 if os.getenv("CHUNK_P_DROP_REFERENCES") is not None: 

337 result[key].setdefault( 

338 "drop_references", _env_bool("CHUNK_P_DROP_REFERENCES") 

339 ) 

340 return result
