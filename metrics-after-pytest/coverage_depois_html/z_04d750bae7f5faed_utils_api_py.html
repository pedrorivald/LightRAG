79# ========== Token Renewal Rate Limiting ========== 

80# Cache to track last renewal time per user (username as key) 

81# Format: {username: last_renewal_timestamp} 

82# 

83# Bounded on purpose (CWE-770, GHSA-3wg5-5w54-3rfm). The key is the JWT "sub" 

84# claim, which is attacker-chosen in any profile running on the default 

85# guest-mode JWT secret, so an unbounded dict here is a memory-exhaustion 

86# primitive. Three independent bounds now apply: the claim itself is capped at 

87# MAX_TOKEN_SUBJECT_LENGTH by AuthHandler.validate_token, only authenticated 

88# requests reach _record_token_renewal (see _renew_token_if_needed), and the 

89# table is bounded below. 

90# 

91# Note the contrast with LoginRateLimiter, the sibling control on this same 

92# request path: it must never evict a live record, because evicting an active 

93# lockout would clear it. Here a "live" entry only suppresses an early renewal, 

94# so evicting one costs a single extra token mint and carries no security 

95# consequence. The cap can therefore be enforced unconditionally, which is what 

96# keeps the table genuinely bounded instead of merely capped-then-full. 

97_token_renewal_cache: "OrderedDict[str, float]" = OrderedDict() 

98_RENEWAL_MIN_INTERVAL = 60 # Minimum 60 seconds between renewals for same user 

99_MAX_TRACKED_RENEWALS = 10_000 # Hard ceiling on distinct tracked usernames 

100 

101# ========== Token Renewal Path Exclusions ========== 

102# Paths that should NOT trigger token auto-renewal 

103# - /health: Health check endpoint, no login required 

104# - /documents/paginated: Client polls this frequently (5-30s), renewal not needed 

105# - /documents/pipeline_status: Client polls this very frequently (2s), renewal not needed 

106_TOKEN_RENEWAL_SKIP_PATHS = [ 

107 "/health", 

108 "/documents/paginated", 

109 "/documents/pipeline_status", 

110]

113def _record_token_renewal(username: str, renewed_at: float) -> None: 

114 """Record a renewal timestamp, keeping ``_token_renewal_cache`` bounded. 

115 

116 Insertion order is maintained as time order, so the oldest entries sit at the 

117 head: entries older than ``_RENEWAL_MIN_INTERVAL`` can no longer influence any 

118 decision and are dropped, then ``_MAX_TRACKED_RENEWALS`` is enforced as a hard 

119 ceiling. Both passes are amortized O(1). See the declaration for why evicting 

120 a still-live entry is safe here. 

121 """ 

122 # Purge dead entries from the head. A backwards clock step (time.time() is not 

123 # monotonic) makes the delta negative and simply stops the purge early; the 

124 # hard ceiling below is the backstop. 

125 while _token_renewal_cache: 

126 oldest_username, oldest_at = next(iter(_token_renewal_cache.items())) 

127 if renewed_at - oldest_at < _RENEWAL_MIN_INTERVAL: 

128 break 

129 del _token_renewal_cache[oldest_username] 

130 

131 # Re-insert at the tail so insertion order stays time order even when a 

132 # username is refreshed. 

133 _token_renewal_cache.pop(username, None) 

134 _token_renewal_cache[username] = renewed_at 

135 

136 while len(_token_renewal_cache) > _MAX_TRACKED_RENEWALS: 

137 _token_renewal_cache.popitem(last=False)

277def get_route_path(scope: Mapping[str, Any], mount_prefix: str = "") -> str: 

278 """The route path a matcher must compare against: ``scope["path"]`` minus the 

279 ASGI mount prefix. 

280 

281 In canonical ASGI form ``scope["path"]`` *includes* ``root_path``, so a 

282 request routed to ``/documents`` on a server mounted at ``/site01`` carries 

283 ``path == "/site01/documents"``. Anything comparing against configured route 

284 paths (``WHITELIST_PATHS``, ``ADMISSION_PATHS``, 

285 ``_TOKEN_RENEWAL_SKIP_PATHS``) must subtract the prefix first, or it compares 

286 two different kinds of path. 

287 

288 ``root_path`` is authoritative when present: ``FastAPI.__call__`` writes it 

289 from ``app.root_path`` before the middleware stack runs, so every layer sees 

290 it. ``mount_prefix`` is the fallback for callers whose scope was not built by 

291 FastAPI (hand-rolled ASGI scopes in tests, for instance). 

292 

293 The subtraction is deliberately guarded rather than an unconditional slice, 

294 because the two callers do not see the same scope shape. The admission 

295 middleware runs *outside* ``_RootPathNormalizationMiddleware`` (that ordering 

296 is intentional, so CORS can wrap its refusals), so in a proxy-strip 

297 deployment it sees a bare ``/documents/upload`` while ``root_path`` is 

298 already ``/site01``. An unguarded ``path[len(prefix):]`` would silently 

299 mangle that into ``ments``. 

300 

301 Mirrors ``starlette._utils.get_route_path`` — inlined rather than imported 

302 because that module is private — with one documented divergence: an 

303 exact-prefix hit returns ``"/"`` where Starlette returns ``""``. Configured 

304 paths are always written with a leading slash, so ``"/"`` is the normal form 

305 the matchers here expect. 

306 """ 

307 path = scope.get("path") or "" 

308 prefix = (scope.get("root_path") or mount_prefix or "").rstrip("/") 

309 if not prefix or not path.startswith(prefix): 

310 return path 

311 remainder = path[len(prefix) :] 

312 if not remainder: 

313 return "/" 

314 if remainder.startswith("/"): 

315 return remainder 

316 # Mid-segment collision (prefix ``/api`` against path ``/apikeys``): the 

317 # prefix is not actually a path-segment prefix here, so nothing is removed. 

318 return path

321def path_is_whitelisted(scope: Mapping[str, Any], *, mount_prefix: str = "") -> bool: 

322 """Whether ``WHITELIST_PATHS`` exempts this request from authentication. 

323 

324 Shared so every layer that has to answer "may this request skip auth?" uses 

325 one matcher: the enforcing route dependency 

326 (:func:`get_combined_auth_dependency`) and the pure-ASGI admission 

327 middleware, which must pre-authenticate before reading a request body and 

328 would otherwise 401 a path the route itself lets through (LR2 §9.3). 

329 

330 Takes the ASGI scope, not a path string, so the mount-prefix normalization 

331 cannot be forgotten at a call site. ``WHITELIST_PATHS`` entries are route 

332 paths written without ``LIGHTRAG_API_PREFIX``; matching the raw 

333 ``request.url.path`` against them made the shipped default ``/health,/api/*`` 

334 exempt *every* route as soon as the mount prefix itself started with 

335 ``/api`` — the whole API, unauthenticated, including ``DELETE /documents``. 

336 

337 A ``/*`` entry matches on path-segment boundaries only, so ``/api/*`` exempts 

338 ``/api`` and everything under ``/api/`` but not a sibling route that merely 

339 starts with those characters (``/graph/*`` must not exempt ``/graphs``). The 

340 catch-all ``/*`` compiles to an empty prefix and still matches everything, 

341 since every path starts with ``/``. 

342 """ 

343 route = get_route_path(scope, mount_prefix) 

344 for pattern, is_prefix in whitelist_patterns: 

345 if is_prefix: 

346 if route == pattern or route.startswith(pattern + "/"): 

347 return True 

348 elif route == pattern: 

349 return True 

350 return False

546def get_auth_status_dependency(api_key: Optional[str] = None): 

547 """Create a dependency that reports whether the request carries accepted 

548 credentials, WITHOUT enforcing authentication (it never raises). 

549 

550 Used by endpoints such as ``/health`` that must stay reachable for 

551 unauthenticated liveness probes (always HTTP 200) while only revealing 

552 sensitive configuration to authenticated callers. The acceptance rules 

553 mirror ``get_combined_auth_dependency`` exactly: 

554 

555 - fully open (no AUTH_ACCOUNTS, no API key): nothing is protected 

556 anywhere, so the request is treated as authenticated. 

557 - password auth (AUTH_ACCOUNTS set): a valid non-guest token, or a 

558 valid API key when one is configured, authenticates. 

559 - API-key-only (API key set, no AUTH_ACCOUNTS): only a valid API key

590def whitelist_exposes_api_routes(whitelist_paths: str) -> bool: 

591 """Return True if WHITELIST_PATHS exempts any Ollama-compatible /api route. 

592 

593 Mirrors the prefix/exact matching in get_combined_auth_dependency so that a 

594 catch-all entry such as ``/*`` (which strips to an empty prefix and matches 

595 every request path, including ``/api/chat``) is recognized as exposing the 

596 /api routes — not just literal ``/api...`` entries. 

597 """ 

598 for entry in whitelist_paths.split(","): 

599 entry = entry.strip() 

600 if not entry: 

601 continue 

602 if entry.endswith("/*"): 

603 # Prefix match: this entry exempts an /api route when some /api path 

604 # starts with the prefix ("/api".startswith(prefix) also covers the 

605 # empty catch-all prefix from "/*") or the prefix is itself under 

606 # /api/. The "/api/" boundary matters: "/apiary/*" only exempts 

607 # /apiary..., not /api/chat, so it must NOT be flagged. 

608 prefix = entry[:-2] 

609 if "/api".startswith(prefix) or prefix.startswith("/api/"): 

610 return True 

611 else: 

612 # Exact match: only the literal path is exempted. 

613 if entry == "/api" or entry.startswith("/api/"): 

614 return True 

615 return False
