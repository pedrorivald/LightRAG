1"""No-op format handlers for the two "nothing to parse" document formats. 

2 

3These are internal (``user_selectable=False``) parsers dispatched by *format*, 

4not by a user engine choice: 

5 

6- :class:`ReuseParser` handles ``lightrag`` rows — a document already parsed 

7 by some engine whose sidecar exists. Reached only on resume/retry (a doc 

8 that finished parsing but failed/interrupted at a later stage and is pulled 

9 back through the parse queue). Re-uses the stored content + sidecar 

10 instead of re-parsing (the original source may already be archived). 

11- :class:`PassthroughParser` handles ``raw`` rows — content supplied verbatim 

12 at insert time (e.g. ``ainsert`` of a plain string). 

13 

14Both mirror the corresponding branches of the former ``parse_native`` exactly 

15(no disk writes, ``parse_stage_skipped=True``, no ``parse_engine`` key). 

16""" 

17 

18from __future__ import annotations 

19 

20from lightrag.constants import FULL_DOCS_FORMAT_LIGHTRAG, FULL_DOCS_FORMAT_RAW 

21from lightrag.parser.base import BaseParser, ParseContext, ParseResult 

22 

23 

24class ReuseParser(BaseParser): 

25 """Reuse an already-parsed (``lightrag``-format) document's sidecar.""" 

26 

27 engine_name = "reuse" 

28 

29 async def parse(self, ctx: ParseContext) -> ParseResult: 

30 from lightrag.utils_pipeline import ( 

31 sidecar_blocks_path, 

32 strip_lightrag_doc_prefix, 

33 ) 

34 

35 doc_format = ctx.content_data.get("parse_format", FULL_DOCS_FORMAT_LIGHTRAG) 

36 merged_text = strip_lightrag_doc_prefix( 

37 ctx.content_data.get("content"), doc_format 

38 ) 

39 # ``sidecar_location`` may be absent on historical/abnormal rows; tolerate 

40 # it (blocks_path="") rather than failing or re-routing to extraction. 

41 blocks_path = ( 

42 sidecar_blocks_path(ctx.content_data.get("sidecar_location")) or "" 

43 ) 

44 return ParseResult( 

45 doc_id=ctx.doc_id, 

46 file_path=ctx.file_path, 

47 parse_format=doc_format, 

48 content=merged_text, 

49 blocks_path=blocks_path, 

50 parse_stage_skipped=True, 

51 ) 

52 

53 

54class PassthroughParser(BaseParser): 

55 """Pass ``raw``-format content through verbatim (no parser ran).""" 

56 

57 engine_name = "passthrough" 

58 

59 async def parse(self, ctx: ParseContext) -> ParseResult: 

60 return ParseResult( 

61 doc_id=ctx.doc_id, 

62 file_path=ctx.file_path, 

63 parse_format=FULL_DOCS_FORMAT_RAW, 

64 content=ctx.content_data.get("content", ""), 

65 blocks_path="", 

66 parse_stage_skipped=True, 

67 )
