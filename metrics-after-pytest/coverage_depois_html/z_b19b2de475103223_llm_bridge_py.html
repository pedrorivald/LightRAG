1"""Synchronous LLM bridge for parser extract hooks running in worker threads. 

2 

3Native parser ``extract`` hooks are synchronous and run in a worker thread, 

4while the LLM role funcs (and their cache) are async and bound to the 

5LightRAG owning loop. :class:`SyncLLMBridge` crosses that boundary: it is 

6constructed ON the loop thread (capturing the loop), and called FROM the 

7worker thread, where it submits the coroutine via 

8``run_coroutine_threadsafe`` and waits with a short-interval poll so a 

9cancellation can interrupt the wait promptly. 

10 

11The poll timeout is NOT an LLM timeout — total LLM timeout/retry semantics 

12belong to the role wrapper (``llm_timeout`` etc.); the bridge waits 

13indefinitely, checking its cancel events once per interval. Engine-agnostic 

14on purpose: future xlsx/pptx native engines reuse it as-is. 

15""" 

16 

17from __future__ import annotations 

18 

19import asyncio 

20import concurrent.futures 

21import threading 

22from collections.abc import Callable, Coroutine 

23from typing import Any 

24 

25from lightrag.parser.exceptions import ( 

26 ParseCancelled, 

27 first_cancellation, 

28 normalize_cancel_events, 

29) 

30 

31 

32class SyncLLMBridge: 

33 """Call an async LLM submit function from a synchronous worker thread. 

34 

35 Args: 

36 loop: The event loop the coroutine must run on. Capture it on the 

37 loop thread (``asyncio.get_running_loop()``) before entering the 

38 worker thread. 

39 submit: Async callable ``(prompt, *, system_prompt=None) -> str`` 

40 executed on ``loop`` (typically wrapping 

41 ``use_llm_func_with_cache``). 

42 cancel_events: Events polled between waits. An entry may be an Event, 

43 or an ``(Event, exception_type)`` pair when callers need the 

44 cancellation source preserved. Any set event aborts the wait after 

45 at most one poll interval. 

46 poll_interval: Seconds per ``future.result`` wait slice. 

47 """ 

48 

49 def __init__( 

50 self, 

51 loop: asyncio.AbstractEventLoop, 

52 submit: Callable[..., Coroutine[Any, Any, str]], 

53 *, 

54 cancel_events: tuple[ 

55 threading.Event | tuple[threading.Event, type[ParseCancelled]], ... 

56 ] = (), 

57 poll_interval: float = 1.0, 

58 ) -> None: 

59 self._loop = loop 

60 self._submit = submit 

61 self._cancel_events = normalize_cancel_events(cancel_events) 

62 self._poll_interval = max(0.01, float(poll_interval)) 

63 # The loop thread id at construction time. Calling the bridge from 

64 # that thread would block the loop the coroutine needs — a guaranteed 

65 # deadlock — so __call__ turns it into an immediate error. 

66 self._loop_thread_id = threading.get_ident() 

67 

68 def _cancellation_exception(self, message: str) -> ParseCancelled | None: 

69 return first_cancellation(self._cancel_events, message) 

70 

71 def __call__(self, prompt: str, *, system_prompt: str | None = None) -> str: 

72 if threading.get_ident() == self._loop_thread_id: 

73 raise RuntimeError( 

74 "SyncLLMBridge called from the event-loop thread; this would " 

75 "deadlock waiting on the loop it is blocking. Call it only " 

76 "from a worker thread (parser extract runs in one)." 

77 ) 

78 cancellation = self._cancellation_exception( 

79 "parse cancelled before the LLM call" 

80 ) 

81 if cancellation is not None: 

82 raise cancellation 

83 future = asyncio.run_coroutine_threadsafe( 

84 self._submit(prompt, system_prompt=system_prompt), self._loop 

85 ) 

86 while True: 

87 cancellation = self._cancellation_exception( 

88 "parse cancelled while awaiting the LLM" 

89 ) 

90 if cancellation is not None: 

91 # Propagates cancellation into the loop-side coroutine chain; 

92 # whether the underlying provider request truly aborts is up 

93 # to its implementation — the worker thread exits regardless. 

94 if not future.cancel() and future.done(): 94 ↛ 98line 94 didn't jump to line 98 because the condition on line 94 was never true

95 # Already finished (perhaps with an exception): consume the 

96 # result so asyncio does not log "exception was never 

97 # retrieved" for the abandoned future. 

98 try: 

99 future.exception() 

100 except Exception: 

101 pass 

102 raise cancellation 

103 try: 

104 return future.result(timeout=self._poll_interval) 

105 except concurrent.futures.TimeoutError: 

106 continue 

107 except concurrent.futures.CancelledError as exc: 

108 raise ParseCancelled("LLM call cancelled") from exc
